   115→/** 'first_blood' → 'First blood' (fallback for titles) */
   116→function prettyName(code: string): string {
   117→  const s = code.split('_').slice(1).join(' ').trim();
   118→  return s ? s.charAt(0).toUpperCase() + s.slice(1) : code;
   119→}

   121→/** Localized achievement label built from its metric + goal (readable in all 4 locales). */
   122→function achLabel(a: { id: string; metric: string; goal: number }, t: (k: TKey) => string): string {
   123→  const g = a.goal;
   124→  switch (a.metric) {
   125→    case 'wins': return `${g} × ${t('wins')}`;
   126→    case 'kindWins': return `${g > 1 ? `${g} × ` : ''}${t('quest_q_kind')}`;
   127→    case 'runWins': return `${g > 1 ? `${g} × ` : ''}${t('quest_q_run')}`;
   128→    case 'games': return `${g} × ${t('games_played')}`;
   129→    case 'onlineWins': return `${g} × ${t('online')}`;
   130→    case 'onlineGames': return `${g} × ${t('online')}`;
   131→    case 'hardWins': return `${g} × AI ${t('ai_hard')}`;
   132→    case 'streakBest': return `${g} × ${t('streak')}`;
   133→    case 'itemsOwned': return `${g} × 🛍`;
   134→    case 'level': return `${t('level')} ${g}`;
   135→    case 'questsDone': return `${g} × ${t('daily_quests')}`;
   136→    case 'draws': return `${g} × ${t('draws_stat')}`;
   137→    default: return prettyName(a.id);
   138→  }
   139→}
   140→
   141→/** Resolve an achievement id to its catalog def → localized label. */
   142→function achLabelById(id: string, t: (k: TKey) => string): string {
   143→  const def = ACHIEVEMENTS.find((x) => x.id === id);
   144→  return def ? achLabel(def, t) : prettyName(id);
   145→}

   147→function fmtDate(iso: string, locale: Locale): string {
   148→  try {
   149→    return new Intl.DateTimeFormat(locale, { year: 'numeric', month: 'short', day: 'numeric' }).format(new Date(iso));
   150→  } catch {
   151→    return iso.slice(0, 10);
   152→  }
   153→}
   154→
   155→function timeAgo(iso: string, locale: Locale): string {
   156→  try {
   157→    const rtf = new Intl.RelativeTimeFormat(locale, { numeric: 'auto' });
   158→    const s = Math.round((Date.now() - new Date(iso).getTime()) / 1000);
   159→    if (s < 60) return rtf.format(-Math.max(s, 1), 'second');
   160→    const m = Math.round(s / 60);
   161→    if (m < 60) return rtf.format(-m, 'minute');
   162→    const h = Math.round(m / 60);
   163→    if (h < 24) return rtf.format(-h, 'hour');
   164→    return rtf.format(-Math.round(h / 24), 'day');
   165→  } catch {
   166→    return '';
   167→  }
   168→}

   170→function AchIcon({ name, className }: { name: string; className?: string }) {

   172→  return <Icon className={className} />;
   173→}

   481→              unlocked.map((a) => {
   482→                const checked = showcase.includes(a.id);
   483→                return (
   484→                  <label
   485→                    key={a.id}
   486→                    className="flex cursor-pointer items-center gap-2 rounded-lg px-2 py-1 text-xs transition-colors hover:bg-muted/60"
   487→                  >
   488→                    <Checkbox
   489→                      checked={checked}
   490→                      disabled={!checked && showcase.length >= 3}
   491→                      onCheckedChange={(v) => void toggleShowcase(a.id, v === true)}
   492→                    />
   493→                    <span
   494→                      className="grid h-6 w-6 shrink-0 place-items-center rounded"
   495→                      style={{ background: `${RARITY_META[a.rarity].color}1a`, color: RARITY_META[a.rarity].color }}

   497→                      <AchIcon name={a.icon} className="h-3.5 w-3.5" />
   498→                    </span>
   499→                    <span className="truncate font-bold">{achLabelById(a.id, t)}</span>
   500→                  </label>
   501→                );
   502→              })
   503→            )}
   504→          </div>
   505→        </section>
   506→      </div>
   507→
   508→      {/* ---------- achievements ---------- */}
   509→      <section>
   510→        <h2 className="mb-2 flex items-center gap-2 text-base font-black">
   511→          <Sparkles className="h-4 w-4 text-fuchsia-400" /> {t('achievements')}
   512→        </h2>
   513→        {achs === null ? (
   514→          <div className="grid grid-cols-2 gap-3 sm:grid-cols-3 lg:grid-cols-4">
   515→            {Array.from({ length: 8 }).map((_, i) => (
   516→              <Skeleton key={i} className="h-32 rounded-2xl" />
   517→            ))}
   518→          </div>
   519→        ) : (
   520→          <div className="grid grid-cols-2 gap-3 sm:grid-cols-3 lg:grid-cols-4">
   521→            {achs.map((a) => {
   522→              const meta = RARITY_META[a.rarity];
   523→              const Icon = ACH_ICONS[a.icon] ?? Sparkles;
   524→              return (
   525→                <div
   526→                  key={a.id}
   527→                  className={cn(
   528→                    'royal-card flex flex-col rounded-2xl p-3 transition-all hover:scale-[1.02]',
   529→                    !a.unlockedAt && 'opacity-75',
   530→                    a.unlockedAt && 'glow-gold',
   531→                  )}
   532→                  style={{ borderColor: `${meta.color}44` }}
   533→                >
   534→                  <div className="flex items-start justify-between">
   535→                    <span
   536→                      className="grid h-9 w-9 place-items-center rounded-lg"
   537→                      style={{ background: `${meta.color}1f`, color: meta.color }}
   538→                    >
   539→                      <Icon className="h-5 w-5" />
   540→                    </span>
   541→                    {a.unlockedAt ? (
   542→                      <Badge variant="outline" className="border-emerald-500/40 bg-emerald-500/10 text-[9px] font-black text-emerald-300">
   543→                        ✓ {t('unlocked')}
   544→                      </Badge>

   546→                      <span className="text-[10px] text-muted-foreground" title={t('locked')}>
   547→                        🔒
   548→                      </span>
   549→                    )}
   550→                  </div>
   551→                  <div className="mt-2 truncate text-xs font-black">{achLabelById(a.id, t)}</div>
   552→                  <Progress value={(a.progress / a.goal) * 100} className="mt-2 h-1.5" />
   553→                  <div className="mt-1.5 flex items-center justify-between text-[10px] text-muted-foreground">
   554→                    <span className="font-bold">
   555→                      {localizeDigits(Math.min(a.progress, a.goal), locale)} / {localizeDigits(a.goal, locale)}
   556→                    </span>
   557→                    <span className="font-black text-amber-400">+{localizeDigits(a.coins, locale)} 🪙</span>
   558→                  </div>
   559→                  {a.unlockedAt && <div className="mt-1 text-[9px] text-muted-foreground">📅 {fmtDate(a.unlockedAt, locale)}</div>}
   560→                </div>
   561→              );
   562→            })}
   563→          </div>
   564→        )}
   565→      </section>
   566→
   567→      {/* ---------- match history ---------- */}
   568→      <section className="royal-card rounded-2xl p-4">
   569→        <h3 className="flex items-center gap-2 text-sm font-black">
   570→          <Swords className="h-4 w-4 text-emerald-400" /> {t('history')}
   571→        </h3>
   572→        <div className="nice-scroll mt-3 max-h-80 space-y-2 overflow-y-auto pe-1">
   573→          {games === null ? (
   574→            Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-12 rounded-xl" />)
   575→          ) : games.length === 0 ? (
   576→            <p className="py-6 text-center text-xs text-muted-foreground">{t('no_history')}</p>
   577→          ) : (
   578→            games.map((g) => {
   579→              const mb = modeBadge(g.mode);
   580→              const result = g.draw
   581→                ? { label: t('draws_stat'), cls: 'border-zinc-400/40 bg-zinc-400/10 text-zinc-300' }
   582→                : g.youWon
   583→                  ? { label: t('wins'), cls: 'border-emerald-500/40 bg-emerald-500/10 text-emerald-300' }
   584→                  : { label: t('losses'), cls: 'border-rose-500/40 bg-rose-500/10 text-rose-300' };
   585→              return (
   586→                <div
   587→                  key={g.id}
   588→                  className="flex flex-wrap items-center justify-between gap-2 rounded-xl border border-amber-500/10 bg-muted/30 px-3 py-2 transition-colors hover:bg-muted/50"
   589→                >
   590→                  <div className="flex min-w-0 items-center gap-2">
   591→                    <Badge variant="outline" className={cn('shrink-0 text-[9px] font-black', mb.cls)}>
   592→                      {mb.label}
   593→                    </Badge>
   594→                    <span className="truncate text-xs font-bold">
   595→                      {t('opponent')}: {opponentOf(g)}
   596→                    </span>
   597→                  </div>
   598→                  <div className="flex flex-wrap items-center gap-2 text-[10px] text-muted-foreground">
   599→                    <span>{endLabel(g.endReason)}</span>
   600→                    <span>♟ {localizeDigits(g.moves, locale)}</span>
   601→                    <span>⏱ {timeAgo(g.createdAt, locale)}</span>
   602→                    <Badge variant="outline" className={cn('text-[9px] font-black', result.cls)}>
   603→                      {result.label}
   604→                    </Badge>
   605→                  </div>
   606→                </div>
   607→              );
   608→            })
   609→          )}
   610→        </div>
   611→      </section>
   612→
   613→      {/* ---------- settings ---------- */}
   614→      <section className="royal-card rounded-2xl p-4">
   615→        <h3 className="flex items-center gap-2 text-sm font-black">
   616→          <Settings2 className="h-4 w-4 text-sky-400" /> {t('settings')}
   617→        </h3>
   618→        <div className="mt-3 flex flex-wrap items-center gap-x-6 gap-y-3">
   619→          <div className="flex items-center gap-2">
   620→            <Switch
   621→              id="pf-sound"
   622→              checked={user.soundOn}
   623→              onCheckedChange={(v) => {
   624→                patchUser({ soundOn: v });
   625→                setSoundEnabled(v);
   626→                sfx.click();
   627→                api.patch('/api/profile', { soundOn: v }).catch(() => undefined);
   628→              }}
   629→            />
   630→            <Label htmlFor="pf-sound" className="cursor-pointer text-xs font-bold">
   631→              <Volume2 className="me-1 inline h-3.5 w-3.5" />
   632→              {t('sound')}
   633→            </Label>
   634→          </div>
   635→          <div className="flex items-center gap-2">
   636→            <Globe className="h-4 w-4 text-muted-foreground" />
   637→            <Select
   638→              value={locale}
   639→              onValueChange={(v) => {
   640→                setLocale(v as Locale);
   641→                sfx.click();
   642→              }}
   643→            >
   644→              <SelectTrigger size="sm" className="w-36" aria-label={t('language')}>
   645→                <SelectValue />
   646→              </SelectTrigger>
   647→              <SelectContent>
   648→                {LOCALES.map((l) => (
   649→                  <SelectItem key={l} value={l}>
   650→                    {LOCALE_META[l].flag} {LOCALE_META[l].label}
   651→                  </SelectItem>
   652→                ))}
   653→              </SelectContent>
   654→            </Select>
   655→          </div>
   656→        </div>
   657→      </section>
   658→
   659→      {/* ---------- report player ---------- */}
   660→      <section className="royal-card rounded-2xl p-4">
   661→        <Collapsible open={repOpen} onOpenChange={setRepOpen}>
   662→          <div className="flex items-center justify-between gap-2">
   663→            <h3 className="flex items-center gap-2 text-sm font-black">
   664→              <Flag className="h-4 w-4 text-rose-400" /> {t('report')}
   665→            </h3>
   666→            <CollapsibleTrigger asChild>
   667→              <Button variant="ghost" size="sm" aria-label={t('report')}>
   668→                <ChevronDown className={cn('h-4 w-4 text-muted-foreground transition-transform', repOpen && 'rotate-180')} />
   669→              </Button>
   670→            </CollapsibleTrigger>
   671→          </div>
   672→          <CollapsibleContent className="mt-3 space-y-2">
   673→            <div className="flex flex-wrap gap-2">
   674→              <Input
   675→                dir="ltr"
   676→                className="min-w-40 flex-1"
   677→                placeholder={t('reported_player')}
   678→                value={repTarget}
   679→                maxLength={32}
   680→                onChange={(e) => setRepTarget(e.target.value)}
   681→              />
   682→              <Select value={repReason} onValueChange={(v) => setRepReason(v as (typeof REASONS)[number])}>
   683→                <SelectTrigger size="sm" className="w-44" aria-label={t('reason')}>
   684→                  <SelectValue />
   685→                </SelectTrigger>
   686→                <SelectContent>
   687→                  {REASONS.map((r) => (
   688→                    <SelectItem key={r} value={r}>
   689→                      {t(`report_reason_${r}` as TKey)}
   690→                    </SelectItem>
   691→                  ))}
   692→                </SelectContent>
   693→              </Select>
   694→            </div>
   695→            <Textarea
   696→              rows={2}
   697→              maxLength={500}
   698→              value={repDetails}
   699→              onChange={(e) => setRepDetails(e.target.value)}
   700→              placeholder={t('reason')}
   701→            />
   702→            <div className="flex justify-end">
   703→              <Button
   704→                size="sm"
   705→                variant="outline"
   706→                className="border-rose-500/40 text-rose-300 hover:bg-rose-500/10"
   707→                disabled={repBusy || !repTarget.trim()}

   710→                <Flag className="h-4 w-4" /> {t('report')}
