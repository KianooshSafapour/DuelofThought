    94→    94→/** Immutably write the equipped code into the right skin/avatar/title field. */
    95→    95→function equipApply<T extends { pieceSkin: string; boardSkin: string; background: string; effectSkin: string; avatarItem: string; titleItem: string | null }>(
    96→    96→  obj: T,
    97→    97→  field: EquipField,
    98→    98→  code: string,
    99→    99→): T {
   100→   100→  const next = { ...obj };
   101→   101→  if (field === 'pieceSkin') next.pieceSkin = code;
   102→   102→  else if (field === 'boardSkin') next.boardSkin = code;
   103→   103→  else if (field === 'background') next.background = code;
   104→   104→  else if (field === 'effectSkin') next.effectSkin = code;
   105→   105→  else if (field === 'avatarItem') next.avatarItem = code;
   106→   106→  else next.titleItem = code;
   107→   107→  return next;
   108→   108→}

   329→   329→/** Lightweight confetti — absolutely-positioned animated divs, no new deps. */
   330→   330→function Confetti({ seed }: { seed: number }) {
   331→   331→  const parts = useMemo(
   332→   332→    () =>
   333→   333→      Array.from({ length: 30 }, (_, i) => ({
   334→   334→        id: i,
   335→   335→        x: (Math.random() - 0.5) * 340,
   336→   336→        y: -(70 + Math.random() * 170),
   337→   337→        rot: Math.random() * 720 - 360,
   338→   338→        dur: 1 + Math.random(),
   339→   339→        delay: Math.random() * 0.18,
   340→   340→        c: CONFETTI_COLORS[i % CONFETTI_COLORS.length],
   341→   341→        w: 5 + Math.random() * 6,
   342→   342→      })),
   343→   343→    [seed],
   344→   344→  );
   345→   345→  return (
   346→   346→    <div className="pointer-events-none absolute inset-0 overflow-hidden" aria-hidden>
   347→   347→      {parts.map((p) => (
   348→   348→        <motion.span
   349→   349→          key={`${seed}-${p.id}`}
   350→   350→          className="absolute left-1/2 top-[38%] block rounded-[2px]"
   351→   351→          style={{ width: p.w, height: Math.round(p.w * 0.5), background: p.c }}
   352→   352→          initial={{ x: 0, y: 0, opacity: 1, rotate: 0 }}
   353→   353→          animate={{ x: p.x, y: [0, p.y, 330], opacity: [1, 1, 0], rotate: p.rot }}
   354→   354→          transition={{ duration: p.dur + 0.9, delay: p.delay, ease: 'easeOut', times: [0, 0.42, 1] }}
   355→   355→        />
   356→   356→      ))}
   357→   357→    </div>
   358→   358→  );
   359→   359→}
