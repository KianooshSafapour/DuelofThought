    50→function mulberry32(seed: number): () => number {
    51→  let s = seed | 0;
    52→  return () => {
    53→    s = (s + 0x6d2b79f5) | 0;
    54→    let z = Math.imul(s ^ (s >>> 15), 1 | s);
    55→    z = (z + Math.imul(z ^ (z >>> 7), 61 | z)) ^ z;
    56→    return ((z ^ (z >>> 14)) >>> 0) / 4294967296;
    57→  };
    58→}

    74→function shade(hex: string, amt: number): string {
    75→  const n = parseInt(hex.slice(1), 16);
    76→  let r = (n >> 16) & 255;
    77→  let g = (n >> 8) & 255;
    78→  let b = n & 255;
    79→  if (amt >= 0) {
    80→    r += (255 - r) * amt; g += (255 - g) * amt; b += (255 - b) * amt;
    81→  } else {
    82→    r *= 1 + amt; g *= 1 + amt; b *= 1 + amt;
    83→  }
    84→  return `rgb(${r | 0},${g | 0},${b | 0})`;
    85→}

    87→function hexA(hex: string, a: number): string {
    88→  const n = parseInt(hex.slice(1), 16);
    89→  return `rgba(${(n >> 16) & 255},${(n >> 8) & 255},${n & 255},${a})`;
    90→}

   128→/** Drag vector → capped, fuel-affordable launch Δv (pure; used by prediction & release). */
   129→function aimDelta(s: Sim): { vx: number; vy: number; dv: number } {
   130→  if (!s.drag) return { vx: 0, vy: 0, dv: 0 };
   131→  const dx = (s.drag.cx - s.drag.sx) * LAUNCH_SCALE;
   132→  const dy = (s.drag.cy - s.drag.sy) * LAUNCH_SCALE;
   133→  const L = Math.hypot(dx, dy);
   134→  const maxDv = Math.min(MAX_DRAG * LAUNCH_SCALE, s.fuel / LAUNCH_FUEL_PER_DV);
   135→  const dv = Math.min(L, maxDv);
   136→  if (L < 1e-6 || dv <= 0) return { vx: 0, vy: 0, dv: 0 };
   137→  return { vx: (dx / L) * dv, vy: (dy / L) * dv, dv };
   138→}

   140→function drawSphere(ctx: CanvasRenderingContext2D, x: number, y: number, r: number, color: string, lx: number, ly: number, arcs: number, t: number): void {
   141→  // soft glow
   142→  ctx.save();
   143→  ctx.shadowColor = hexA(color, 0.55);
   144→  ctx.shadowBlur = 16;
   145→  const g = ctx.createRadialGradient(x + lx * r * 0.45, y + ly * r * 0.45, r * 0.12, x, y, r);
   146→  g.addColorStop(0, shade(color, 0.55));
   147→  g.addColorStop(0.55, color);
   148→  g.addColorStop(1, shade(color, -0.62));
   149→  ctx.fillStyle = g;
   150→  ctx.beginPath();
   151→  ctx.arc(x, y, r, 0, Math.PI * 2);
   152→  ctx.fill();
   153→  ctx.restore();
   154→  // rotating surface arcs (clip to sphere)
   155→  if (arcs > 0) {
   156→    ctx.save();
   157→    ctx.beginPath();
   158→    ctx.arc(x, y, r - 1, 0, Math.PI * 2);
   159→    ctx.clip();
   160→    ctx.strokeStyle = 'rgba(0,0,0,0.18)';
   161→    ctx.lineWidth = Math.max(1.4, r * 0.06);
   162→    for (let i = 0; i < arcs; i++) {
   163→      const a0 = t * (0.14 + i * 0.05) * (i % 2 ? -1 : 1) + i * 2.4;
   164→      ctx.beginPath();
   165→      ctx.arc(x, y, r * (0.4 + i * 0.26), a0, a0 + 1.9 + i * 0.4);
   166→      ctx.stroke();
   167→    }
   168→    ctx.restore();
   169→  }
   170→  // rim light on the lit side
   171→  const la = Math.atan2(ly, lx);
   172→  ctx.strokeStyle = 'rgba(255,255,255,0.30)';
   173→  ctx.lineWidth = 1.6;
   174→  ctx.beginPath();
   175→  ctx.arc(x, y, r - 1.1, la - 1.7, la + 1.7);
   176→  ctx.stroke();
   177→}

   237→function drawTarget(ctx: CanvasRenderingContext2D, x: number, y: number, r: number, t: number): void {
   238→  const p = 0.5 + 0.5 * Math.sin(t * 2.4);
   239→  ctx.save();
   240→  // outer rotating dashed ring
   241→  ctx.setLineDash([14, 10]);
   242→  ctx.lineDashOffset = -t * 26;
   243→  ctx.strokeStyle = `rgba(251,191,36,${0.5 + 0.3 * p})`;
   244→  ctx.lineWidth = 2.6;
   245→  ctx.beginPath();
   246→  ctx.arc(x, y, r, 0, Math.PI * 2);
   247→  ctx.stroke();
   248→  ctx.setLineDash([]);
   249→  // inner ring
   250→  ctx.strokeStyle = 'rgba(251,191,36,0.22)';
   251→  ctx.lineWidth = 1.6;
   252→  ctx.beginPath();
   253→  ctx.arc(x, y, r * 0.62, 0, Math.PI * 2);
   254→  ctx.stroke();
   255→  // rotating compass ticks
   256→  const ta = t * 0.6;
   257→  ctx.strokeStyle = 'rgba(253,230,138,0.8)';
   258→  ctx.lineWidth = 2.4;
   259→  for (let i = 0; i < 4; i++) {
   260→    const a = ta + (i * Math.PI) / 2;
   261→    ctx.beginPath();
   262→    ctx.moveTo(x + Math.cos(a) * (r - 9), y + Math.sin(a) * (r - 9));
   263→    ctx.lineTo(x + Math.cos(a) * (r + 9), y + Math.sin(a) * (r + 9));
   264→    ctx.stroke();
   265→  }
   266→  // core beacon
   267→  ctx.shadowColor = 'rgba(251,191,36,0.9)';
   268→  ctx.shadowBlur = 14 + 10 * p;
   269→  ctx.fillStyle = `rgba(253,230,138,${0.75 + 0.25 * p})`;
   270→  ctx.beginPath();
   271→  ctx.arc(x, y, 6 + 2 * p, 0, Math.PI * 2);
   272→  ctx.fill();
   273→  ctx.restore();
   274→}

   276→function drawPod(ctx: CanvasRenderingContext2D, x: number, y: number, i: number, t: number): void {
   277→  const s = 10 * (1 + 0.16 * Math.sin(t * 3 + i * 1.7));
   278→  ctx.save();
   279→  ctx.translate(x, y);
   280→  ctx.shadowColor = 'rgba(45,212,191,0.9)';
   281→  ctx.shadowBlur = 12;
   282→  ctx.fillStyle = 'rgba(45,212,191,0.92)';
   283→  ctx.beginPath();
   284→  for (let k = 0; k < 6; k++) {
   285→    const a = (k * Math.PI) / 3 - Math.PI / 6 + t * 0.5;
   286→    const px = Math.cos(a) * s;
   287→    const py = Math.sin(a) * s;
   288→    if (k === 0) ctx.moveTo(px, py); else ctx.lineTo(px, py);
   289→  }
   290→  ctx.closePath();
   291→  ctx.fill();
   292→  ctx.shadowBlur = 0;
   293→  ctx.fillStyle = 'rgba(240,253,250,0.95)';
   294→  ctx.beginPath();
   295→  ctx.arc(0, 0, s * 0.3, 0, Math.PI * 2);
   296→  ctx.fill();
   297→  ctx.restore();
   298→}

   302→function OrbitCanvas({ mission, label, onWin, onLose, onHud }: {
   303→  mission: Mission;
   304→  label: string;
   305→  onWin: (pods: number, fuelLeft: number, time: number) => void;
   306→  onLose: (reason: 'crash' | 'lost' | 'timeout') => void;
   307→  onHud: (s: HudSnap) => void;
   308→}) {
   309→  const canvasRef = useRef<HTMLCanvasElement | null>(null);
   310→  const simRef = useRef<Sim>(initSim(mission));
   311→  const viewRef = useRef({ cssW: 900 });
   312→  const cbRef = useRef({ onWin, onLose, onHud });
   313→  const stars = useMemo(() => makeStars(MISSIONS.indexOf(mission) + 3), [mission]);
   314→
   315→  useEffect(() => {
   316→    cbRef.current = { onWin, onLose, onHud };
   317→  }, [onWin, onLose, onHud]);
   318→
   319→  /* keep the backing store sized (device-pixel sharp) */
   320→  useEffect(() => {
   321→    const canvas = canvasRef.current;
   322→    if (!canvas) return;
   323→    const resize = () => {
   324→      const w = canvas.clientWidth || 900;
   325→      const dpr = Math.min(2, window.devicePixelRatio || 1);
   326→      canvas.width = Math.round(w * dpr);
   327→      canvas.height = Math.round(w * 0.75 * dpr);
   328→      viewRef.current.cssW = w;
   329→    };
   330→    resize();
   331→    const ro = new ResizeObserver(resize);
   332→    ro.observe(canvas);
   333→    return () => ro.disconnect();
   334→  }, []);
   335→
   336→  /* physics + render loop */
   337→  useEffect(() => {
   338→    const canvas = canvasRef.current;
   339→    if (!canvas) return;
   340→    const ctx = canvas.getContext('2d');
   341→    if (!ctx) return;
   342→    const sun = mission.bodies.find((b) => b.kind === 'sun');
   343→
   344→    const burst = (x: number, y: number, c: string, n: number, speed: number) => {
   345→      const s = simRef.current;
   346→      for (let i = 0; i < n; i++) {
   347→        const a = Math.random() * Math.PI * 2;
   348→        const v = speed * (0.3 + Math.random() * 0.9);
   349→        s.particles.push({ x, y, vx: Math.cos(a) * v, vy: Math.sin(a) * v, life: 0.7 + Math.random() * 0.7, max: 1.4, c, r: 1.5 + Math.random() * 2.5 });
   350→      }
   351→    };
   352→
   353→    const step = (dt: number) => {
   354→      const s = simRef.current;
   355→      if (s.phase !== 'flight') return;
   356→      s.time += dt;
   357→      s.thrustingNow = false;
   358→      let tx = 0;
   359→      let ty = 0;
   360→      if (s.thrusting && s.fuel > 0) {
   361→        const dx = s.aimX - s.x;
   362→        const dy = s.aimY - s.y;
   363→        const d = Math.hypot(dx, dy);
   364→        if (d > 6) {
   365→          tx = (dx / d) * THRUST_ACCEL;
   366→          ty = (dy / d) * THRUST_ACCEL;
   367→          s.fuel = Math.max(0, s.fuel - FUEL_BURN * dt);
   368→          s.thrustingNow = true;
   369→        }
   370→      }
   371→      stepProbe(s, mission.bodies, dt, tx, ty);
   372→      const speed = Math.hypot(s.vx, s.vy);
   373→      if (speed > 1) s.heading = Math.atan2(s.vy, s.vx);
   374→      s.trailTick = (s.trailTick + 1) % 2;
   375→      if (s.trailTick === 0) {
   376→        s.trail.push({ x: s.x, y: s.y });
   377→        if (s.trail.length > 120) s.trail.shift();
   378→      }
   379→      // science pods
   380→      for (let i = 0; i < mission.pods.length; i++) {
   381→        if (!s.pods[i] && Math.hypot(s.x - mission.pods[i].x, s.y - mission.pods[i].y) < POD_PICKUP) {
   382→          s.pods[i] = true;
   383→          burst(mission.pods[i].x, mission.pods[i].y, '#5eead4', 12, 60);
   384→          sfx.coin();
   385→        }
   386→      }
   387→      // win — capture radius
   388→      if (Math.hypot(s.x - mission.target.x, s.y - mission.target.y) < mission.target.r) {
   389→        s.phase = 'won';
   390→        burst(s.x, s.y, '#fbbf24', 26, 90);
   391→        sfx.win();
   392→        cbRef.current.onWin(s.pods.filter(Boolean).length, s.fuel, s.time);
   393→        return;
   394→      }
   395→      // crash into a surface
   396→      if (hitsBody(s.x, s.y, mission.bodies)) {
   397→        s.phase = 'crashed';
   398→        s.shake = 1;
   399→        burst(s.x, s.y, '#fb923c', 30, 120);
   400→        sfx.bad();
   401→        cbRef.current.onLose('crash');
   402→        return;
   403→      }
   404→      // drifted out of bounds
   405→      if (outOfBounds(s.x, s.y)) {
   406→        s.phase = 'lost';
   407→        sfx.lose();
   408→        cbRef.current.onLose('lost');
   409→        return;
   410→      }
   411→      // mission clock
   412→      if (s.time >= TIME_LIMIT) {
   413→        s.phase = 'timeout';
   414→        sfx.lose();
   415→        cbRef.current.onLose('timeout');
   416→      }
   417→    };
   418→
   419→    const draw = (t: number) => {
   420→      const s = simRef.current;
   421→      const cssW = viewRef.current.cssW;
   422→      const px = WORLD.w / cssW; // css px → world units
   423→      const dpr = canvas.width / Math.max(1, cssW);
   424→      const scale = dpr * (cssW / WORLD.w);
   425→      ctx.setTransform(scale, 0, 0, scale, 0, 0);
   426→      if (s.shake > 0) {
   427→        s.shake = Math.max(0, s.shake - 0.03);
   428→        ctx.translate((Math.random() - 0.5) * s.shake * 14, (Math.random() - 0.5) * s.shake * 14);
   429→      }
   430→      // deep space
   431→      ctx.fillStyle = '#04070d';
   432→      ctx.fillRect(-40, -40, WORLD.w + 80, WORLD.h + 80);
   433→      for (const n of NEBULAS) {
   434→        const g = ctx.createRadialGradient(n.x, n.y, 0, n.x, n.y, n.r);
   435→        g.addColorStop(0, `rgba(${n.c},${n.a})`);

   526→            ctx.lineWidth = pass === 0 ? 5 * f + 1 : 1.6 * f + 0.5;
   527→            ctx.beginPath();
   528→            ctx.moveTo(s.trail[i - 1].x, s.trail[i - 1].y);
   529→            ctx.lineTo(s.trail[i].x, s.trail[i].y);
   530→            ctx.stroke();
   531→          }
   532→        }

   535→      const pr = 15;
   536→      // selection pulse while aiming
   537→      if (s.phase === 'aiming') {
   538→        ctx.strokeStyle = `rgba(94,234,212,${0.5 + 0.3 * Math.sin(t * 2.6)})`;
   539→        ctx.lineWidth = 2;
   540→        ctx.beginPath();
   541→        ctx.arc(s.x, s.y, pr + 7 + 2.5 * Math.sin(t * 2.6), 0, Math.PI * 2);
   542→        ctx.stroke();
   543→      }
   544→      ctx.save();
   545→      ctx.translate(s.x, s.y);
   546→      ctx.rotate(s.heading);
   547→      if (s.thrustingNow) {
   548→        const fl = 15 + Math.random() * 9;
   549→        const fg = ctx.createLinearGradient(-8, 0, -8 - fl, 0);
   550→        fg.addColorStop(0, 'rgba(253,224,71,0.95)');
   551→        fg.addColorStop(0.6, 'rgba(245,158,11,0.55)');
   552→        fg.addColorStop(1, 'rgba(245,158,11,0)');

   556→        ctx.lineTo(-8 - fl, 0);
   557→        ctx.lineTo(-7, -4.5);

   559→        ctx.fill();

   561→      ctx.shadowColor = 'rgba(251,191,36,0.8)';

   567→      ctx.beginPath();

   570→      ctx.lineTo(-4, 0);
   571→      ctx.lineTo(-8, -7);

   637→        const cx = Math.max(26, Math.min(WORLD.w - 26, s.x));

   672→    let raf = 0;

   690→      } else if (s.phase === 'aiming') {

   692→        s.prediction = null;

   696→        const p = s.particles[i];
   697→        p.life -= dtReal;
   698→        p.x += p.vx * dtReal;
   699→        p.y += p.vy * dtReal;
   700→        p.vx *= 0.985;
   701→        p.vy *= 0.985;
   702→        if (p.life <= 0) s.particles.splice(i, 1);

   708→          phase: s.phase,
   709→          fuel: s.fuel,
   710→          fuelMax: mission.startFuel,

   714→          dv: s.dv,
   715→          aiming: s.phase === 'aiming',
   716→          thrusting: s.thrustingNow,
   717→          coasting: s.phase === 'flight' && s.fuel <= 0,
   718→        });
   719→      }
   720→    };
   721→    raf = requestAnimationFrame(loop);
   722→    return () => cancelAnimationFrame(raf);
   723→  }, [mission, stars]);
   724→
   725→  /* pointer input — aiming (drag anywhere) & mid-flight burns */
   726→  const getWorld = (e: React.PointerEvent<HTMLCanvasElement>): { x: number; y: number } => {
   727→    const rect = e.currentTarget.getBoundingClientRect();
   728→    return { x: ((e.clientX - rect.left) / rect.width) * WORLD.w, y: ((e.clientY - rect.top) / rect.height) * WORLD.h };
   729→  };
   730→
   731→  const onPointerDown = (e: React.PointerEvent<HTMLCanvasElement>) => {
   732→    const s = simRef.current;
   733→    const p = getWorld(e);
   734→    try { e.currentTarget.setPointerCapture(e.pointerId); } catch { /* noop */ }
   735→    if (s.phase === 'aiming') {
   736→      s.drag = { sx: p.x, sy: p.y, cx: p.x, cy: p.y };
   737→    } else if (s.phase === 'flight') {
   738→      s.thrusting = true;
   739→      s.aimX = p.x;
   740→      s.aimY = p.y;
   741→    }
   742→  };
   743→
   744→  const onPointerMove = (e: React.PointerEvent<HTMLCanvasElement>) => {
   745→    const s = simRef.current;
   746→    if (!s.drag && !s.thrusting) return;
   747→    const p = getWorld(e);
   748→    if (s.drag) {
   749→      s.drag.cx = p.x;
   750→      s.drag.cy = p.y;
   751→    } else if (s.thrusting && s.phase === 'flight') {
   752→      s.aimX = p.x;
   753→      s.aimY = p.y;
   754→    }
   755→  };
   756→
   757→  const endPointer = (e: React.PointerEvent<HTMLCanvasElement>) => {
   758→    const s = simRef.current;
   759→    if (s.phase === 'aiming' && s.drag) {
   760→      const d = aimDelta(s);
   761→      if (d.dv < 1.2) { // tiny drag = cancel, no fuel spent
   762→        s.drag = null;
   763→        s.dv = 0;
   764→        s.prediction = null;
   765→        return;
   766→      }
   767→      s.vx = d.vx;
   768→      s.vy = d.vy;
   769→      s.fuel = Math.max(0, s.fuel - d.dv * LAUNCH_FUEL_PER_DV);
   770→      s.heading = Math.atan2(d.vy, d.vx);
   771→      s.drag = null;
   772→      s.phase = 'flight';
   773→      sfx.drop();
   774→    } else if (s.thrusting) {
   775→      s.thrusting = false;
   776→      s.thrustingNow = false;
   777→    }
   778→    try { e.currentTarget.releasePointerCapture(e.pointerId); } catch { /* noop */ }
   779→  };
   780→
   781→  return (
   782→    <canvas
   783→      ref={canvasRef}
   784→      className="block aspect-[4/3] w-full touch-none select-none"
   785→      dir="ltr"
   786→      onPointerDown={onPointerDown}
   787→      onPointerMove={onPointerMove}
   788→      onPointerUp={endPointer}
   789→      onPointerCancel={endPointer}
   790→      onContextMenu={(e) => e.preventDefault()}
   791→      aria-label={label}
   792→    />
   793→  );

   813→/* progress: 'doi_orbit_mission' = highest unlocked index; 'doi_orbit_best' = per-mission bests.
   814→   Lazy-loaded (client-only mount — the game opens after a user click, never SSR'd). */
   815→function loadUnlocked(): number {
   816→  if (typeof window === 'undefined') return 0;
   817→  try {
   818→    const u = Number(window.localStorage.getItem('doi_orbit_mission') ?? '0');
   819→    if (Number.isFinite(u)) return Math.min(Math.max(0, Math.floor(u)), MISSIONS.length - 1);
   820→  } catch { /* noop */ }
   821→  return 0;
   822→}
   823→
   824→function loadBests(): number[] {
   825→  const empty = MISSIONS.map(() => 0);
   826→  if (typeof window === 'undefined') return empty;
   827→  try {
   828→    const b = JSON.parse(window.localStorage.getItem('doi_orbit_best') ?? '[]');
   829→    if (Array.isArray(b)) return MISSIONS.map((_, i) => (typeof b[i] === 'number' && b[i] > 0 ? b[i] : 0));
   830→  } catch { /* noop */ }
   831→  return empty;
   832→}

   888→    setResult(null);
   889→    setDlgOpen(true);
