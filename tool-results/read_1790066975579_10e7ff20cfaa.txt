   122→/** 'first_blood' → 'First blood' (fallback for titles) */
   123→function prettyName(code: string): string {
   124→  const s = code.split('_').slice(1).join(' ').trim();
   125→  return s ? s.charAt(0).toUpperCase() + s.slice(1) : code;
   126→}
   127→
   128→/** Localized achievement label built from its metric + goal (readable in all 4 locales). */
   129→function achLabel(a: { id: string; metric: string; goal: number }, t: (k: TKey) => string): string {
   130→  const g = a.goal;
   131→  switch (a.metric) {
   132→    case 'wins': return `${g} × ${t('wins')}`;
   133→    case 'kindWins': return `${g > 1 ? `${g} × ` : ''}${t('quest_q_kind')}`;
   134→    case 'runWins': return `${g > 1 ? `${g} × ` : ''}${t('quest_q_run')}`;
   135→    case 'games': return `${g} × ${t('games_played')}`;
   136→    case 'onlineWins': return `${g} × ${t('online')}`;
   137→    case 'onlineGames': return `${g} × ${t('online')}`;
   138→    case 'hardWins': return `${g} × AI ${t('ai_hard')}`;
   139→    case 'streakBest': return `${g} × ${t('streak')}`;
   140→    case 'itemsOwned': return `${g} × 🛍`;
   141→    case 'level': return `${t('level')} ${g}`;
   142→    case 'questsDone': return `${g} × ${t('daily_quests')}`;
   143→    case 'draws': return `${g} × ${t('draws_stat')}`;
   144→    default: return prettyName(a.id);
   145→  }
   146→}
   147→
   148→/** Resolve an achievement id to its catalog def → localized label. */
   149→function achLabelById(id: string, t: (k: TKey) => string): string {
   150→  const def = ACHIEVEMENTS.find((x) => x.id === id);
   151→  return def ? achLabel(def, t) : prettyName(id);
   152→}

   154→function fmtDate(iso: string, locale: Locale): string {
   155→  try {
   156→    return new Intl.DateTimeFormat(locale, { year: 'numeric', month: 'short', day: 'numeric' }).format(new Date(iso));
   157→  } catch {
   158→    return iso.slice(0, 10);
   159→  }
   160→}
   161→
   162→function timeAgo(iso: string, locale: Locale): string {
   163→  try {
   164→    const rtf = new Intl.RelativeTimeFormat(locale, { numeric: 'auto' });
   165→    const s = Math.round((Date.now() - new Date(iso).getTime()) / 1000);
   166→    if (s < 60) return rtf.format(-Math.max(s, 1), 'second');
   167→    const m = Math.round(s / 60);
   168→    if (m < 60) return rtf.format(-m, 'minute');
   169→    const h = Math.round(m / 60);
   170→    if (h < 24) return rtf.format(-h, 'hour');
   171→    return rtf.format(-Math.round(h / 24), 'day');
   172→  } catch {
   173→    return '';
   174→  }
   175→}

   177→function AchIcon({ name, className }: { name: string; className?: string }) {
   178→  const Icon = ACH_ICONS[name] ?? Sparkles;
   179→  return <Icon className={className} />;
   180→}

   483→          <div className="nice-scroll mt-3 max-h-40 space-y-1 overflow-y-auto pe-1">
   484→            {achs === null ? (
   485→              <Skeleton className="h-16 w-full" />
   486→            ) : unlocked.length === 0 ? (
   487→              <p className="py-2 text-xs text-muted-foreground">—</p>
   488→            ) : (
   489→              unlocked.map((a) => {
   490→                const checked = showcase.includes(a.id);
   491→                return (
   492→                  <label
   493→                    key={a.id}
   494→                    className="flex cursor-pointer items-center gap-2 rounded-lg px-2 py-1 text-xs transition-colors hover:bg-muted/60"
   495→                  >
   496→                    <Checkbox
   497→                      checked={checked}
   498→                      disabled={!checked && showcase.length >= 3}
   499→                      onCheckedChange={(v) => void toggleShowcase(a.id, v === true)}
   500→                    />
   501→                    <span
   502→                      className="grid h-6 w-6 shrink-0 place-items-center rounded"
   503→                      style={{ background: `${RARITY_META[a.rarity].color}1a`, color: RARITY_META[a.rarity].color }}
   504→                    >
   505→                      <AchIcon name={a.icon} className="h-3.5 w-3.5" />
   506→                    </span>
   507→                    <span className="truncate font-bold">{achLabelById(a.id, t)}</span>
   508→                  </label>
   509→                );
   510→              })
   511→            )}
   512→          </div>
   513→        </section>
   514→      </div>
   515→
   516→      {/* ---------- achievements ---------- */}
   517→      <section>
   518→        <h2 className="mb-2 flex items-center gap-2 text-base font-black">
   519→          <Sparkles className="h-4 w-4 text-fuchsia-400" /> {t('achievements')}
   520→        </h2>
   521→        {achs === null ? (
   522→          <div className="grid grid-cols-2 gap-3 sm:grid-cols-3 lg:grid-cols-4">
   523→            {Array.from({ length: 8 }).map((_, i) => (
   524→              <Skeleton key={i} className="h-32 rounded-2xl" />
   525→            ))}
   526→          </div>
   527→        ) : (
   528→          <div className="grid grid-cols-2 gap-3 sm:grid-cols-3 lg:grid-cols-4">
   529→            {achs.map((a) => {
   530→              const meta = RARITY_META[a.rarity];
   531→              const Icon = ACH_ICONS[a.icon] ?? Sparkles;
   532→              return (
   533→                <div
   534→                  key={a.id}
   535→                  className={cn(
   536→                    'royal-card flex flex-col rounded-2xl p-3 transition-all hover:scale-[1.02]',
   537→                    !a.unlockedAt && 'opacity-75',
   538→                    a.unlockedAt && 'glow-gold',
   539→                  )}
   540→                  style={{ borderColor: `${meta.color}44` }}
   541→                >
   542→                  <div className="flex items-start justify-between">
   543→                    <span
   544→                      className="grid h-9 w-9 place-items-center rounded-lg"
   545→                      style={{ background: `${meta.color}1f`, color: meta.color }}
   546→                    >
   547→                      <Icon className="h-5 w-5" />
   548→                    </span>
   549→                    {a.unlockedAt ? (
   550→                      <Badge variant="outline" className="border-emerald-500/40 bg-emerald-500/10 text-[9px] font-black text-emerald-300">
   551→                        ✓ {t('unlocked')}
   552→                      </Badge>
   553→                    ) : (
   554→                      <span className="text-[10px] text-muted-foreground" title={t('locked')}>
   555→                        🔒
   556→                      </span>
   557→                    )}
   558→                  </div>
   559→                  <div className="mt-2 truncate text-xs font-black">{achLabelById(a.id, t)}</div>
   560→                  <Progress value={(a.progress / a.goal) * 100} className="mt-2 h-1.5" />
   561→                  <div className="mt-1.5 flex items-center justify-between text-[10px] text-muted-foreground">
   562→                    <span className="font-bold">
   563→                      {localizeDigits(Math.min(a.progress, a.goal), locale)} / {localizeDigits(a.goal, locale)}
   564→                    </span>
   565→                    <span className="font-black text-amber-400">+{localizeDigits(a.coins, locale)} 🪙</span>
   566→                  </div>
   567→                  {a.unlockedAt && <div className="mt-1 text-[9px] text-muted-foreground">📅 {fmtDate(a.unlockedAt, locale)}</div>}
   568→                </div>
   569→              );
   570→            })}
   571→          </div>
   572→        )}
   573→      </section>
   574→
   575→      {/* ---------- match history ---------- */}
   576→      <section className="royal-card rounded-2xl p-4">
   577→        <h3 className="flex items-center gap-2 text-sm font-black">
   578→          <Swords className="h-4 w-4 text-emerald-400" /> {t('history')}
   579→        </h3>
   580→        <div className="nice-scroll mt-3 max-h-80 space-y-2 overflow-y-auto pe-1">
   581→          {games === null ? (
   582→            Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-12 rounded-xl" />)
   583→          ) : games.length === 0 ? (
   584→            <p className="py-6 text-center text-xs text-muted-foreground">{t('no_history')}</p>
   585→          ) : (
   586→            games.map((g) => {
   587→              const mb = modeBadge(g.mode);
   588→              const result = g.draw
   589→                ? { label: t('draws_stat'), cls: 'border-zinc-400/40 bg-zinc-400/10 text-zinc-300' }
   590→                : g.youWon
   591→                  ? { label: t('wins'), cls: 'border-emerald-500/40 bg-emerald-500/10 text-emerald-300' }
   592→                  : { label: t('losses'), cls: 'border-rose-500/40 bg-rose-500/10 text-rose-300' };
   593→              return (
   594→                <div
   595→                  key={g.id}
   596→                  className="flex flex-wrap items-center justify-between gap-2 rounded-xl border border-amber-500/10 bg-muted/30 px-3 py-2 transition-colors hover:bg-muted/50"
   597→                >
   598→                  <div className="flex min-w-0 items-center gap-2">
   599→                    <Badge variant="outline" className={cn('shrink-0 text-[9px] font-black', mb.cls)}>
   600→                      {mb.label}
   601→                    </Badge>
   602→                    <span className="truncate text-xs font-bold">
   603→                      {t('opponent')}: {opponentOf(g)}
   604→                    </span>
   605→                  </div>
   606→                  <div className="flex flex-wrap items-center gap-2 text-[10px] text-muted-foreground">
   607→                    <span>{endLabel(g.endReason)}</span>
   608→                    <span>♟ {localizeDigits(g.moves, locale)}</span>
   609→                    <span>⏱ {timeAgo(g.createdAt, locale)}</span>
   610→                    <Badge variant="outline" className={cn('text-[9px] font-black', result.cls)}>
   611→                      {result.label}
   612→                    </Badge>
   613→                    <button
   614→                      onClick={() => { setReplayId(g.id); sfx.click(); }}
   615→                      className="inline-flex items-center gap-1 rounded-lg border border-amber-500/40 bg-amber-500/10 px-2 py-1 text-[10px] font-black text-amber-300 transition-colors hover:bg-amber-500/20"
   616→                      aria-label={t('replay')}
   617→                    >
   618→                      <Film className="h-3 w-3" /> {t('replay')}
   619→                    </button>
   620→                  </div>
   621→                </div>
   622→              );
   623→            })
   624→          )}
   625→        </div>
   626→      </section>
   627→
   628→      {/* ---------- replay viewer (remounts per game) ---------- */}
   629→      {replayId && <ReplayModal gameId={replayId} onOpenChange={(o) => !o && setReplayId(null)} />}
   630→
   631→      {/* ---------- settings ---------- */}
   632→      <section className="royal-card rounded-2xl p-4">
   633→        <h3 className="flex items-center gap-2 text-sm font-black">
   634→          <Settings2 className="h-4 w-4 text-sky-400" /> {t('settings')}
   635→        </h3>
   636→        <div className="mt-3 flex flex-wrap items-center gap-x-6 gap-y-3">
   637→          <div className="flex items-center gap-2">
   638→            <Switch
   639→              id="pf-sound"
   640→              checked={user.soundOn}
   641→              onCheckedChange={(v) => {
   642→                patchUser({ soundOn: v });
   643→                setSoundEnabled(v);
   644→                sfx.click();
   645→                api.patch('/api/profile', { soundOn: v }).catch(() => undefined);
   646→              }}
   647→            />
   648→            <Label htmlFor="pf-sound" className="cursor-pointer text-xs font-bold">
   649→              <Volume2 className="me-1 inline h-3.5 w-3.5" />
   650→              {t('sound')}
   651→            </Label>
   652→          </div>
   653→          <div className="flex items-center gap-2">
   654→            <MusicSwitch />
   655→            <Label htmlFor="pf-music" className="cursor-pointer text-xs font-bold">
   656→              <Music2 className="me-1 inline h-3.5 w-3.5" />
   657→              {t('music')}
   658→            </Label>
   659→          </div>
   660→          <MoodPicker />
   661→          <div className="flex items-center gap-2">
   662→            <Globe className="h-4 w-4 text-muted-foreground" />
   663→            <Select
   664→              value={locale}
   665→              onValueChange={(v) => {

   671→                <SelectValue />

   673→              <SelectContent>
   674→                {LOCALES.map((l) => (
   675→                  <SelectItem key={l} value={l}>
   676→                    {LOCALE_META[l].flag} {LOCALE_META[l].label}

   684→
   685→      {/* ---------- report player ---------- */}
   686→      <section className="royal-card rounded-2xl p-4">
   687→        <Collapsible open={repOpen} onOpenChange={setRepOpen}>
   688→          <div className="flex items-center justify-between gap-2">
   689→            <h3 className="flex items-center gap-2 text-sm font-black">
   690→              <Flag className="h-4 w-4 text-rose-400" /> {t('report')}
   691→            </h3>
   692→            <CollapsibleTrigger asChild>
   693→              <Button variant="ghost" size="sm" aria-label={t('report')}>
   694→                <ChevronDown className={cn('h-4 w-4 text-muted-foreground transition-transform', repOpen && 'rotate-180')} />
   695→              </Button>
   696→            </CollapsibleTrigger>
   697→          </div>
   698→          <CollapsibleContent className="mt-3 space-y-2">
   699→            <div className="flex flex-wrap gap-2">
   700→              <Input
   701→                dir="ltr"
   702→                className="min-w-40 flex-1"

   728→            <div className="flex justify-end">

   730→                size="sm"

   742→    </div>
