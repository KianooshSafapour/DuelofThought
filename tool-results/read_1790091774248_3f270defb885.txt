   123→const clamp = (v: number, a: number, b: number) => Math.max(a, Math.min(b, v));

   125→function angLerp(a: number, b: number, k: number) {
   126→  const d = ((b - a + Math.PI * 3) % (Math.PI * 2)) - Math.PI;
   127→  return a + d * k;
   128→}
   129→
   130→/** Circle vs axis-aligned rect: push-out normal + penetration (null = miss). */
   131→function circleRect(px: number, py: number, r: number, w: MMWall) {
   132→  const cx = clamp(px, w.x, w.x + w.w);
   133→  const cy = clamp(py, w.y, w.y + w.h);
   134→  const dx = px - cx;
   135→  const dy = py - cy;
   136→  const d2 = dx * dx + dy * dy;
   137→  if (d2 >= r * r) return null;
   138→  if (d2 > 1e-6) {
   139→    const d = Math.sqrt(d2);
   140→    return { nx: dx / d, ny: dy / d, pen: r - d };
   141→  }
   142→  const l = px - w.x;
   143→  const rt = w.x + w.w - px;
   144→  const t = py - w.y;
   145→  const b = w.y + w.h - py;
   146→  const m = Math.min(l, rt, t, b);
   147→  if (m === l) return { nx: -1, ny: 0, pen: r + l };
   148→  if (m === rt) return { nx: 1, ny: 0, pen: r + rt };
   149→  if (m === t) return { nx: 0, ny: -1, pen: r + t };
   150→  return { nx: 0, ny: 1, pen: r + b };
   151→}

   153→/** Liang–Barsky segment vs AABB — used for drone line-of-sight. */
   154→function segHitsRect(x1: number, y1: number, x2: number, y2: number, w: MMWall) {
   155→  let t0 = 0;
   156→  let t1 = 1;
   157→  const dx = x2 - x1;
   158→  const dy = y2 - y1;
   159→  const p = [-dx, dx, -dy, dy];
   160→  const q = [x1 - w.x, w.x + w.w - x1, y1 - w.y, w.y + w.h - y1];
   161→  for (let i = 0; i < 4; i++) {
   162→    if (p[i] === 0) {
   163→      if (q[i] < 0) return false;
   164→      continue;
   165→    }
   166→    const r = q[i] / p[i];
   167→    if (p[i] < 0) {
   168→      if (r > t1) return false;
   169→      if (r > t0) t0 = r;
   170→    } else {
   171→      if (r < t0) return false;
   172→      if (r < t1) t1 = r;
   173→    }
   174→  }
   175→  return true;
   176→}
   177→
   178→function hasLOS(x1: number, y1: number, x2: number, y2: number, walls: MMWall[]) {
   179→  for (const w of walls) if (segHitsRect(x1, y1, x2, y2, w)) return false;
   180→  return true;
   181→}
   182→
   183→function pointFree(x: number, y: number, r: number, walls: MMWall[]) {
   184→  for (const w of walls) if (circleRect(x, y, r, w)) return false;
   185→  return true;
   186→}

   188→/** Push a circle body out of every wall; damp velocity into surfaces (e = bounce). */
   189→function resolveWalls(x: number, y: number, r: number, vx: number, vy: number, e: number, walls: MMWall[]) {
   190→  for (const w of walls) {
   191→    const c = circleRect(x, y, r, w);
   192→    if (!c) continue;
   193→    x += c.nx * c.pen;
   194→    y += c.ny * c.pen;
   195→    const vn = vx * c.nx + vy * c.ny;
   196→    if (vn < 0) {
   197→      vx -= (1 + e) * vn * c.nx;
   198→      vy -= (1 + e) * vn * c.ny;
   199→    }
   200→  }
   201→  return { x, y, vx, vy };
   202→}
   203→
   204→/** Keep a body inside the arena frame; e = 1 gives a clean pulse bounce. */
   205→function resolveBounds(x: number, y: number, r: number, vx: number, vy: number, e: number) {
   206→  let hit = false;
   207→  if (x < PAD + r) { x = PAD + r; if (vx < 0) { vx = -vx * e; hit = true; } }
   208→  if (x > ARENA_W - PAD - r) { x = ARENA_W - PAD - r; if (vx > 0) { vx = -vx * e; hit = true; } }
   209→  if (y < PAD + r) { y = PAD + r; if (vy < 0) { vy = -vy * e; hit = true; } }
   210→  if (y > ARENA_H - PAD - r) { y = ARENA_H - PAD - r; if (vy > 0) { vy = -vy * e; hit = true; } }
   211→  return { x, y, vx, vy, hit };
   212→}

   214→/** Intercept angle at a moving target for a projectile of `speed` (per frame). */
   215→function interceptAngle(sx: number, sy: number, tx: number, ty: number, tvx: number, tvy: number, speed: number) {
   216→  const dx = tx - sx;
   217→  const dy = ty - sy;
   218→  const a = tvx * tvx + tvy * tvy - speed * speed;
   219→  const b = 2 * (dx * tvx + dy * tvy);
   220→  const c = dx * dx + dy * dy;
   221→  let t = -1;
   222→  if (Math.abs(a) < 1e-4) {
   223→    if (Math.abs(b) > 1e-4) t = -c / b;
   224→  } else {
   225→    const disc = b * b - 4 * a * c;
   226→    if (disc >= 0) {
   227→      const sq = Math.sqrt(disc);
   228→      const t1 = (-b - sq) / (2 * a);
   229→      const t2 = (-b + sq) / (2 * a);
   230→      t = Math.min(t1, t2);
   231→      if (t < 0) t = Math.max(t1, t2);
   232→    }
   233→  }
   234→  if (!Number.isFinite(t) || t < 0) t = Math.hypot(dx, dy) / speed;
   235→  t = Math.min(t, 400); // sanity bound; real intercepts are far below this
   236→  return Math.atan2(dy + tvy * t, dx + tvx * t);
   237→}

   239→function rr(ctx: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, r: number) {
   240→  const rad = Math.min(r, w / 2, h / 2);
   241→  ctx.beginPath();
   242→  ctx.moveTo(x + rad, y);
   243→  ctx.arcTo(x + w, y, x + w, y + h, rad);
   244→  ctx.arcTo(x + w, y + h, x, y + h, rad);
   245→  ctx.arcTo(x, y + h, x, y, rad);
   246→  ctx.arcTo(x, y, x + w, y, rad);
   247→  ctx.closePath();
   248→}

   602→    /* ---------------- update: drones (state machine) ---------------- */
   603→    const newWaypoint = (g: Game, d: Drone) => {

   612→    const updateDrone = (g: Game, d: Drone, step: number, dt: number) => {
   613→      if (d.state === 'spawn') {
   614→        d.spawnT -= dt;
   615→        d.spin += 0.12 * step;
   616→        if (d.spawnT <= 0) {
   617→          d.state = 'patrol'; // materialize: becomes vulnerable + starts hunting
   618→          newWaypoint(g, d);
   619→        }
   620→        return;
   621→      }
   622→      const spec = droneSpec(d.tier);
   623→      d.flashT = Math.max(0, d.flashT - dt);
   624→      d.dodgeT = Math.max(0, d.dodgeT - dt);
   625→      d.spin += (0.03 + d.tier * 0.004) * step;
   626→
   627→      const pdx = g.px - d.x;
   628→      const pdy = g.py - d.y;
   629→      const pd = Math.hypot(pdx, pdy) || 1;
   630→      const los = hasLOS(d.x, d.y, g.px, g.py, g.walls);
   631→
   632→      /* --- transitions --- */
   633→      if (d.state === 'patrol') {
   634→        if (los && pd < AGGRO) d.state = 'chase';
   635→      } else if (d.state === 'chase') {
   636→        if (!los && pd > AGGRO) {
   637→          d.state = 'patrol';
   638→          newWaypoint(g, d);
   639→        } else if (d.fireT <= 0 && los && pd < FIRE_RANGE) {
   640→          d.state = 'aim';
   641→          d.aimT = spec.aimMs;
   642→          d.losLost = 0;
   643→        }
   644→      } else if (d.state === 'aim') {
   645→        if (!los) {
   646→          d.losLost += dt;
   647→          if (d.losLost > 320) { d.state = 'chase'; d.fireT = 420; }
   648→        } else {
   649→          d.aimT -= dt;
   650→          if (d.aimT <= 0) {
   651→            const tLead = (pd / spec.plasma) * spec.lead;
   652→            const ax = g.px + g.pvx * tLead;
   653→            const ay = g.py + g.pvy * tLead;
   654→            const ang = Math.atan2(ay - d.y, ax - d.x);
   655→            const mx = d.x + Math.cos(ang) * (DRONE_R + 7);
   656→            const my = d.y + Math.sin(ang) * (DRONE_R + 7);
   657→            g.pulses.push({ x: mx, y: my, vx: Math.cos(ang) * spec.plasma, vy: Math.sin(ang) * spec.plasma, r: PLASMA_R, life: 5200, bounces: 0, friendly: false, color: DRONE_COLORS[d.tier - 1] });
   658→            burst(g, mx, my, DRONE_COLORS[d.tier - 1], 4, 1.4);
   659→            d.fireT = spec.fireMs * (0.85 + Math.random() * 0.35);
   660→            d.state = 'chase';
   661→          }
   662→        }
   663→      }
   664→      if (d.state !== 'aim') d.fireT -= dt;
   665→
   666→      /* --- shot-dodge strafe (tier 6+) --- */
   667→      if (spec.dodge && d.dodgeT <= 0) {
   668→        for (const p of g.pulses) {
   669→          if (!p.friendly) continue;
   670→          const ddx = d.x - p.x;
   671→          const ddy = d.y - p.y;
   672→          const dist = Math.hypot(ddx, ddy);
   673→          if (dist > DODGE_RANGE || dist < 1) continue;
   674→          const ux = p.vx / PULSE_SPEED;
   675→          const uy = p.vy / PULSE_SPEED;
   676→          const closing = ux * ddx + uy * ddy;
   677→          const perp = Math.abs(ux * ddy - uy * ddx);
   678→          if (closing > 0 && perp < DODGE_PERP) {
   679→            const pa = Math.atan2(p.vy, p.vx);
   680→            d.dodgeAng = pa + (ux * ddy - uy * ddx > 0 ? 1 : -1) * (Math.PI / 2);
   681→            d.dodgeT = 340;
   682→            break;
   683→          }
   684→        }
   685→      }
   686→
   687→      /* --- steering --- */
   688→      let tx = 0;
   689→      let ty = 0;
   690→      let mul = 1;
   691→      if (d.state === 'patrol') {
   692→        const wdx = d.wp.x - d.x;
   693→        const wdy = d.wp.y - d.y;
   694→        const wd = Math.hypot(wdx, wdy);
   695→        if (wd < 34) newWaypoint(g, d);
   696→        else { tx = wdx / wd; ty = wdy / wd; mul = 0.6; }
   697→      } else if (d.state === 'chase') {
   698→        const ux = pdx / pd;
   699→        const uy = pdy / pd;
   700→        if (!los) {
   701→          // lost sight → push toward the player but bias sideways so we slide
   702→          // around cover instead of grinding into a concave corner (T-junctions)
   703→          const side = d.orbitDir * 0.7;
   704→          const ca = Math.cos(side);
   705→          const sa = Math.sin(side);
   706→          tx = ux * ca - uy * sa;
   707→          ty = ux * sa + uy * ca;
   708→        } else if (pd > KEEP + 60) { tx = ux; ty = uy; }
   709→        else if (pd < KEEP - 70) { tx = -ux; ty = -uy; }
   710→        else { tx = -uy * d.orbitDir; ty = ux * d.orbitDir; }
   711→        if (Math.random() < 0.004 * step) d.orbitDir *= -1;
   712→      } else {
   713→        mul = 0.12; // aiming: dig in
   714→      }
   715→      if (d.dodgeT > 0) { tx += Math.cos(d.dodgeAng) * 1.5; ty += Math.sin(d.dodgeAng) * 1.5; }
   716→
   717→      const tm = Math.hypot(tx, ty);
   718→      let vx = d.vx;
   719→      let vy = d.vy;
   720→      if (tm > 0.01) {
   721→        const sp = spec.speed * mul;
   722→        const ke = Math.min(1, 0.14 * step);
   723→        vx += ((tx / tm) * sp - vx) * ke;
   724→        vy += ((ty / tm) * sp - vy) * ke;
   725→      } else {
   726→        vx *= Math.pow(0.9, step);
   727→        vy *= Math.pow(0.9, step);
   728→      }
   729→      let x = d.x + vx * step;
   730→      let y = d.y + vy * step;
   731→      const res = resolveWalls(x, y, DRONE_R, vx, vy, 0.1, g.walls);
   732→      x = res.x; y = res.y; vx = res.vx; vy = res.vy;
   733→      const b = resolveBounds(x, y, DRONE_R, vx, vy, 0);
   734→      d.x = b.x; d.y = b.y; d.vx = b.vx; d.vy = b.vy;
   735→
   736→      /* anti-stall: wedged against cover (velocity ≈ 0 while trying to move)
   737→         → drop to patrol and walk to a nearby free waypoint to break out */
   738→      if (d.state === 'chase' || d.state === 'patrol') {
   739→        const want = spec.speed * (d.state === 'chase' ? 1 : 0.6);
   740→        if (Math.hypot(d.vx, d.vy) < want * 0.4) d.stuckT += dt;
   741→        else d.stuckT = Math.max(0, d.stuckT - dt * 2);
   742→        if (d.stuckT > 900) {
   743→          d.stuckT = 0;
   744→          d.state = 'patrol';
   745→          let fled = false;
   746→          for (let i = 0; i < 12; i++) {
   747→            const a = Math.random() * Math.PI * 2;
   748→            const rad = 60 + Math.random() * 180;
   749→            const fx = clamp(d.x + Math.cos(a) * rad, PAD + 30, ARENA_W - PAD - 30);
   750→            const fy = clamp(d.y + Math.sin(a) * rad, PAD + 30, ARENA_H - PAD - 30);
   751→            if (pointFree(fx, fy, DRONE_R + 6, g.walls)) { d.wp = { x: fx, y: fy }; fled = true; break; }
   752→          }
   753→          if (!fled) newWaypoint(g, d);
   754→        }
   755→      } else {
   756→        d.stuckT = 0;
   757→      }
   758→
   759→      if (Math.hypot(d.vx, d.vy) > 0.25) d.bodyAng = angLerp(d.bodyAng, Math.atan2(d.vy, d.vx), Math.min(1, 0.16 * step));
   760→
   761→      /* gun tracks the player with tier-scaled lead */
   762→      if (los) {
   763→        const tLead = (pd / spec.plasma) * spec.lead;
   764→        const ax = g.px + g.pvx * tLead;
   765→        const ay = g.py + g.pvy * tLead;
   766→        d.gunAng = angLerp(d.gunAng, Math.atan2(ay - d.y, ax - d.x), Math.min(1, (0.07 + d.tier * 0.02) * step));
   767→      } else if (tm > 0.01) {
   768→        d.gunAng = angLerp(d.gunAng, Math.atan2(vy, vx), Math.min(1, 0.08 * step));
   769→      }
   770→    };

   772→    /* ---------------- update: pulses (banked logic + plasma) ---------------- */
   773→    const updatePulses = (g: Game, step: number, dt: number) => {
   774→      for (let i = g.pulses.length - 1; i >= 0; i--) {
   775→        const p = g.pulses[i];
   776→        p.x += p.vx * step;
   777→        p.y += p.vy * step;

   784→
   785→        /* walls */
   786→        for (const w of g.walls) {
   787→          const c = circleRect(p.x, p.y, p.r, w);
   788→          if (!c) continue;
   789→          p.x += c.nx * c.pen;
   790→          p.y += c.ny * c.pen;
   791→          const vn = p.vx * c.nx + p.vy * c.ny;
   792→          if (vn < 0) {
   793→            p.vx -= 2 * vn * c.nx;
   794→            p.vy -= 2 * vn * c.ny;
   795→            bounced = true;
   796→          }
   797→          break;
   798→        }
   799→
   800→        if (bounced) {
   801→          p.bounces += 1;
   802→          burst(g, p.x, p.y, p.friendly ? '#fde68a' : p.color, 4, 1.4);
   803→          if (p.bounces > PULSE_BOUNCES) {
   804→            burst(g, p.x, p.y, p.friendly ? '#fbbf24' : p.color, 6, 1.6);
   805→            g.pulses.splice(i, 1);
   806→            continue;
   807→          }
   808→          if (p.friendly) {
   809→            const now = performance.now();
   810→            if (now - lastBounceSfx > 60) { sfx.click(); lastBounceSfx = now; }
   811→          }
   812→        }
   813→
   814→        if (p.life <= 0) { g.pulses.splice(i, 1); continue; }
   815→
   816→        if (p.friendly) {
   817→          for (let j = g.drones.length - 1; j >= 0; j--) {
   818→            const d = g.drones[j];
   819→            if (d.state === 'spawn') continue;
   820→            if (Math.hypot(d.x - p.x, d.y - p.y) < DRONE_R + p.r) {
   821→              g.hits += 1;
   822→              d.hp -= 1;
   823→              d.flashT = 130;
   824→              d.vx += p.vx * 0.3;
   825→              d.vy += p.vy * 0.3;
   826→              burst(g, p.x, p.y, DRONE_COLORS[d.tier - 1], 8, 2.2);
   827→              g.pulses.splice(i, 1);
   828→              if (d.hp <= 0) deactivateDrone(g, d, j);
   829→              else sfx.chat();
   830→              break;
   831→            }
   832→          }
   833→        } else if (g.invT <= 0 && Math.hypot(g.px - p.x, g.py - p.y) < PLAYER_R + p.r) {
   834→          g.pulses.splice(i, 1);
   835→          hitPlayer(g);
   836→        }
   837→      }
   838→    };

   863→            pop(g, p.x, p.y - 14, `+♥`, '#34d399');

   877→      const ds = g.drones;
   878→      for (let i = 0; i < ds.length; i++) {
   879→        for (let j = i + 1; j < ds.length; j++) {
   880→          const a = ds[i];
   881→          const b = ds[j];
   882→          const dx = b.x - a.x;
   883→          const dy = b.y - a.y;
   884→          const d = Math.hypot(dx, dy);
   885→          const min = DRONE_R * 2 - 2;
   886→          if (d < min && d > 0.01) {
   887→            const push = (min - d) / 2;
   888→            const nx = dx / d;
   889→            const ny = dy / d;
   890→            a.x -= nx * push; a.y -= ny * push;
   891→            b.x += nx * push; b.y += ny * push;
   892→          } else if (d <= 0.01) {
   893→            a.x -= 1; b.x += 1;
   894→          }
   895→        }
   896→        const d = ds[i];
   897→        const dx = d.x - g.px;
   898→        const dy = d.y - g.py;
   899→        const dist = Math.hypot(dx, dy);
   900→        const min = DRONE_R + PLAYER_R;
   901→        if (dist < min && dist > 0.01) {
   902→          const push = (min - dist) / 2;
   903→          const nx = dx / dist;
   904→          const ny = dy / dist;
   905→          d.x += nx * push; d.y += ny * push;
   906→          g.px -= nx * push; g.py -= ny * push;
   907→        }
   908→      }

   913→      const g = gRef.current;
   914→      if (!g) return;
   915→      g.time += dt;
   916→
   917→      /* particles & popups always settle (nice during banners / game over) */
   918→      for (let i = g.parts.length - 1; i >= 0; i--) {
   919→        const p = g.parts[i];

   928→        const p = g.pops[i];
   929→        p.y -= 0.45 * step;
   930→        p.life -= dt;
   931→        if (p.life <= 0) g.pops.splice(i, 1);

   937→          setBanner(null);
   938→          if (g.pendingVictory) endRun(true);
   939→          else startWave();
   940→        }
   941→        return;
   942→      }
   943→      if (g.phase === 'waveclear') {
   944→        g.nextWaveT -= dt;
   945→        if (g.nextWaveT <= 0) {
   946→          const { t: tt, locale: lc } = useApp.getState();
   947→          if (g.wave >= WAVES.length) {
   948→            g.score += SCORE_VICTORY;

   954→            showBanner(tt('mazemind_victory'), `+${localizeDigits(SCORE_VICTORY, lc)}`);

   958→            g.phase = 'banner';
   959→            setPhase('banner');
   960→            g.bannerT = 1500;
   961→            showBanner(`${tt('mazemind_wave')} ${localizeDigits(g.wave, lc)}`);
   962→          }
   963→        }
   964→        return;
   965→      }
   966→      if (g.phase !== 'playing') return;
   967→
   968→      updatePlayer(g, step, dt);
   969→      updatePulses(g, step, dt);
   970→      for (const d of g.drones) updateDrone(g, d, step, dt);
   971→      separate(g);
   972→      for (const d of g.drones) {
   973→        const res = resolveWalls(d.x, d.y, DRONE_R, d.vx, d.vy, 0, g.walls);
   974→        d.x = res.x; d.y = res.y; d.vx = res.vx; d.vy = res.vy;
   975→      }
   976→      updatePickups(g, step, dt);
   977→
   978→      /* wave cleared? */
   979→      if (g.drones.length === 0) {
   980→        const { t: tt, locale: lc } = useApp.getState();
   981→        g.phase = 'waveclear';
   982→        setPhase('waveclear');
   983→        g.score += SCORE_WAVE;

   985→        if (!g.hitThisWave) {
   986→          g.score += SCORE_PERFECT;
   987→          pop(g, ARENA_W / 2, ARENA_H / 2, `${tt('mazemind_perfect')} +${localizeDigits(SCORE_PERFECT, lc)}`, '#34d399');
   988→        }
   989→        setScore(g.score);
   990→        sfx.win();
