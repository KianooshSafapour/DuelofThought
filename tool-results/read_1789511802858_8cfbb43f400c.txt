    94→/** Small data-fetching hook with refresh + loading + error toast.
    95→ *  `loading` is derived from the last *completed* request (url+tick identity),
    96→ *  so no setState is needed synchronously inside the effect. */
    97→function useAdminData<T>(url: string) {
    98→  const t = useApp((s) => s.t);
    99→  const [done, setDone] = useState<{ url: string; tick: number; data: T | null } | null>(null);
   100→  const [tick, setTick] = useState(0);
   101→
   102→  useEffect(() => {
   103→    let alive = true;
   104→    api.get<T>(url)
   105→      .then((d) => { if (alive) setDone({ url, tick, data: d }); })
   106→      .catch(() => {
   107→        if (alive) {
   108→          setDone({ url, tick, data: null });
   109→          toast.error(t('error'));
   110→        }
   111→      });
   112→    return () => { alive = false; };
   113→  }, [url, tick, t]);
   114→
   115→  const current = done !== null && done.url === url && done.tick === tick;
   116→  return {
   117→    data: current ? done.data : null,
   118→    loading: !current,
   119→    refresh: () => setTick((x) => x + 1),
   120→    /** optimistic in-place patch of the current dataset */
   121→    patch: (fn: (d: T) => T) =>
   122→      setDone((s) => (s !== null && s.url === url && s.data !== null ? { ...s, data: fn(s.data) } : s)),
   123→  };
   124→}

   126→function fmtDate(iso: string | null | undefined, locale: string): string {
   127→  if (!iso) return '—';
   128→  const d = new Date(iso);
   129→  if (Number.isNaN(d.getTime())) return '—';
   130→  return d.toLocaleDateString(locale);
   131→}

   133→function fmtDateTime(iso: string | null | undefined, locale: string): string {
   134→  if (!iso) return '—';
   135→  const d = new Date(iso);
   136→  if (Number.isNaN(d.getTime())) return '—';
   137→  return d.toLocaleString(locale);
   138→}

   140→function apiErrMsg(e: unknown, t: (k: TKey) => string): string {
   141→  const code = (e as Error)?.message ?? '';
   142→  if (code === 'ADMIN_TARGET') return 'این هدف یک مدیر است · Target is an admin';
   143→  if (code === 'SELF') return 'این عمل روی خودتان ممکن نیست · Cannot target yourself';
   144→  if (code === 'RATE') return 'سرعت اقدامات زیاد است · Too many actions, slow down';
   145→  if (code === 'FORBIDDEN') return t('admin_only');
   146→  return t('error');
   147→}

  1327→function StoreTab() {
  1328→  const { t, locale } = useApp();
  1329→  const fmt = (n: number) => localizeDigits(n, locale);
  1330→  const { data, patch, loading, refresh } = useAdminData<{ items: ItemRow[] }>('/api/admin/items');
  1331→  const [drafts, setDrafts] = useState<Record<string, string>>({});
  1332→  const items = data?.items ?? [];
  1333→
  1334→  const savePrice = async (it: ItemRow) => {
  1335→    const raw = drafts[it.code];
  1336→    if (raw === undefined || !Number.isFinite(Number(raw))) return;
  1337→    const price = Math.max(0, Math.floor(Number(raw)));
  1338→    try {
  1339→      await api.post('/api/admin/items', { code: it.code, price });
  1340→      patch((d) => ({ ...d, items: d.items.map((x) => (x.code === it.code ? { ...x, price } : x)) }));
  1341→      setDrafts((dr) => { const n = { ...dr }; delete n[it.code]; return n; });
  1342→      toast.success(`${it.code} — ${t('success')}`);
  1343→    } catch {
  1344→      toast.error(t('error'));
  1345→    }
  1346→  };
  1347→
  1348→  const toggleActive = async (it: ItemRow, active: boolean) => {
  1349→    patch((d) => ({ ...d, items: d.items.map((x) => (x.code === it.code ? { ...x, active } : x)) }));
  1350→    try {
  1351→      await api.post('/api/admin/items', { code: it.code, active });
  1352→      toast.success(t('success'));
  1353→    } catch {
  1354→      toast.error(t('error'));
  1355→      refresh();
  1356→    }
  1357→  };
  1358→
  1359→  return loading && !data ? <ListSkeleton rows={6} /> : (
  1360→    <div className="royal-card rounded-2xl">
  1361→      <div className="overflow-x-auto nice-scroll">
  1362→        <Table className="min-w-[760px]">
  1363→          <TableHeader>
  1364→            <TableRow className="border-amber-500/15 hover:bg-transparent">
  1365→              <TableHead>کد · Code</TableHead>
  1366→              <TableHead>دسته · Category</TableHead>
  1367→              <TableHead>کمیابی · Rarity</TableHead>
  1368→              <TableHead>قیمت · Price</TableHead>
  1369→              <TableHead>فعال · Active</TableHead>
  1370→              <TableHead>خریداران · Buyers</TableHead>
  1371→            </TableRow>
  1372→          </TableHeader>
  1373→          <TableBody>
  1374→            {items.map((it) => {
  1375→              const raw = drafts[it.code];
  1376→              const changed = raw !== undefined && raw !== String(it.price) && Number.isFinite(Number(raw)) && Number(raw) >= 0;
  1377→              return (
  1378→                <TableRow key={it.code} className="border-amber-500/10">
  1379→                  <TableCell className="font-bold" dir="ltr">{it.code}</TableCell>
  1380→                  <TableCell className="text-xs text-muted-foreground">{t(`cat_${it.category}` as TKey)}</TableCell>
  1381→                  <TableCell>
  1382→                    <Badge variant="outline" className={RARITY_STYLE[it.rarity] ?? ''}>{t(`rarity_${it.rarity}` as TKey)}</Badge>
  1383→                  </TableCell>
  1384→                  <TableCell>
  1385→                    <div className="flex items-center gap-1.5">
  1386→                      <Input
  1387→                        dir="ltr"
  1388→                        type="number"
  1389→                        min={0}
  1390→                        max={100000}
  1391→                        className="h-8 w-24"
  1392→                        value={raw ?? String(it.price)}
  1393→                        onChange={(e) => setDrafts((d) => ({ ...d, [it.code]: e.target.value }))}
  1394→                        aria-label={`price ${it.code}`}
  1395→                      />
  1396→                      <Button
  1397→                        variant="ghost"
  1398→                        size="icon"
  1399→                        className="h-8 w-8 text-emerald-300 hover:bg-emerald-500/10 hover:text-emerald-200"

  1415→            {items.length === 0 && (
  1416→              <TableRow>
