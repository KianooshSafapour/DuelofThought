   157→function angLerp(a: number, b: number, k: number) {
   158→  const d = ((b - a + Math.PI * 3) % (Math.PI * 2)) - Math.PI;
   159→  return a + d * k;
   160→}

   162→function circleRect(px: number, py: number, r: number, w: TTWall) {
   163→  const cx = clamp(px, w.x, w.x + w.w);
   164→  const cy = clamp(py, w.y, w.y + w.h);
   165→  const dx = px - cx;
   166→  const dy = py - cy;
   167→  const d2 = dx * dx + dy * dy;
   168→  if (d2 >= r * r) return null;
   169→  if (d2 > 1e-6) {
   170→    const d = Math.sqrt(d2);
   171→    return { nx: dx / d, ny: dy / d, pen: r - d };
   172→  }
   173→  const l = px - w.x;
   174→  const rt = w.x + w.w - px;
   175→  const t = py - w.y;
   176→  const b = w.y + w.h - py;
   177→  const m = Math.min(l, rt, t, b);
   178→  if (m === l) return { nx: -1, ny: 0, pen: r + l };
   179→  if (m === rt) return { nx: 1, ny: 0, pen: r + rt };
   180→  if (m === t) return { nx: 0, ny: -1, pen: r + t };
   181→  return { nx: 0, ny: 1, pen: r + b };
   182→}

   184→function segHitsRect(x1: number, y1: number, x2: number, y2: number, w: TTWall) {
   185→  let t0 = 0;
   186→  let t1 = 1;
   187→  const dx = x2 - x1;
   188→  const dy = y2 - y1;
   189→  const p = [-dx, dx, -dy, dy];
   190→  const q = [x1 - w.x, w.x + w.w - x1, y1 - w.y, w.y + w.h - y1];
   191→  for (let i = 0; i < 4; i++) {
   192→    if (p[i] === 0) {
   193→      if (q[i] < 0) return false;
   194→      continue;
   195→    }
   196→    const r = q[i] / p[i];
   197→    if (p[i] < 0) {
   198→      if (r > t1) return false;
   199→      if (r > t0) t0 = r;
   200→    } else {
   201→      if (r < t0) return false;
   202→      if (r < t1) t1 = r;
   203→    }
   204→  }
   205→  return true;
   206→}

   208→function hasLOS(x1: number, y1: number, x2: number, y2: number, walls: TTWall[]) {
   209→  for (const w of walls) if (segHitsRect(x1, y1, x2, y2, w)) return false;
   210→  return true;
   211→}

   213→function pointFree(x: number, y: number, r: number, walls: TTWall[]) {
   214→  for (const w of walls) if (circleRect(x, y, r, w)) return false;
   215→  return true;
   216→}

   218→function resolveWalls(x: number, y: number, r: number, vx: number, vy: number, e: number, walls: TTWall[]) {
   219→  for (const w of walls) {
   220→    const c = circleRect(x, y, r, w);
   221→    if (!c) continue;
   222→    x += c.nx * c.pen;
   223→    y += c.ny * c.pen;
   224→    const vn = vx * c.nx + vy * c.ny;
   225→    if (vn < 0) {
   226→      vx -= (1 + e) * vn * c.nx;
   227→      vy -= (1 + e) * vn * c.ny;
   228→    }
   229→  }
   230→  return { x, y, vx, vy };
   231→}

   233→function resolveBounds(x: number, y: number, r: number, vx: number, vy: number, e: number) {
   234→  let hit = false;
   235→  if (x < PAD + r) { x = PAD + r; if (vx < 0) { vx = -vx * e; hit = true; } }
   236→  if (x > WORLD_W - PAD - r) { x = WORLD_W - PAD - r; if (vx > 0) { vx = -vx * e; hit = true; } }
   237→  if (y < PAD + r) { y = PAD + r; if (vy < 0) { vy = -vy * e; hit = true; } }
   238→  if (y > WORLD_H - PAD - r) { y = WORLD_H - PAD - r; if (vy > 0) { vy = -vy * e; hit = true; } }
   239→  return { x, y, vx, vy, hit };
   240→}

   242→function interceptAngle(sx: number, sy: number, tx: number, ty: number, tvx: number, tvy: number, speed: number) {
   243→  const dx = tx - sx;
   244→  const dy = ty - sy;
   245→  const a = tvx * tvx + tvy * tvy - speed * speed;
   246→  const b = 2 * (dx * tvx + dy * tvy);
   247→  const c = dx * dx + dy * dy;
   248→  let t = -1;
   249→  if (Math.abs(a) < 1e-4) {
   250→    if (Math.abs(b) > 1e-4) t = -c / b;
   251→  } else {
   252→    const disc = b * b - 4 * a * c;
   253→    if (disc >= 0) {
   254→      const sq = Math.sqrt(disc);
   255→      const t1 = (-b - sq) / (2 * a);
   256→      const t2 = (-b + sq) / (2 * a);
   257→      t = Math.min(t1, t2);
   258→      if (t < 0) t = Math.max(t1, t2);
   259→    }
   260→  }
   261→  if (!Number.isFinite(t) || t < 0) t = Math.hypot(dx, dy) / speed;
   262→  t = Math.min(t, 400);
   263→  return Math.atan2(dy + tvy * t, dx + tvx * t);
   264→}

   266→function rr(ctx: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, r: number) {
   267→  const rad = Math.min(r, w / 2, h / 2);
   268→  ctx.beginPath();
   269→  ctx.moveTo(x + rad, y);
   270→  ctx.arcTo(x + w, y, x + w, y + h, rad);
   271→  ctx.arcTo(x + w, y + h, x, y + h, rad);
   272→  ctx.arcTo(x, y + h, x, y, rad);
   273→  ctx.arcTo(x, y, x + w, y, rad);
   274→  ctx.closePath();
   275→}

   279→function computeParams(tank: TankSpec, upgrades: Record<string, number>): TankParams {
   280→  const p = effStat(tank, 'power', upgrades);
   281→  const rate = effStat(tank, 'rate', upgrades);
   282→  const spd = effStat(tank, 'speed', upgrades);
   283→  const arm = effStat(tank, 'armor', upgrades);
   284→  const en = effStat(tank, 'energy', upgrades);
   285→  return {
   286→    damage: 1 + Math.floor(p / 3.2),
   287→    fireCd: Math.round(380 - rate * 20),
   288→    speed: 2.55 + spd * 0.2,
   289→    maxHp: 2 + Math.round(arm),
   290→    energyRate: 3.6 + en * 0.55,
   291→    chain: tank.id === 'tesla',
   292→    splash: tank.id === 'volcano',
   293→    pierce: tank.id === 'prism',
   294→    blink: tank.id === 'mirage',
   295→    regen: tank.id === 'omega',
   296→    special: tank.special,
   297→  };
   298→}

   304→function droneKindFor(m: MissionDef, i: number): DroneKind {
   305→  const rng = mulberry32(m.seed * 31 + i * 7);
   306→  const roll = rng();
   307→  if (m.level >= 3 && roll < 0.22) return 'sniper';
   308→  if (roll < 0.5) return 'scout';
   309→  return 'gunner';
   310→}

   465→    setBanner(null);
