    86→function prettyName(code: string): string {
    87→  const s = code.split('_').slice(1).join(' ').trim();
    88→  return s ? s.charAt(0).toUpperCase() + s.slice(1) : code;
    89→}

    94→/** Immutably write the equipped code into the right skin/avatar/title field. */
    95→function equipApply<T extends { pieceSkin: string; boardSkin: string; background: string; effectSkin: string; avatarItem: string; titleItem: string | null }>(
    96→  obj: T,
    97→  field: EquipField,
    98→  code: string,
    99→): T {
   100→  const next = { ...obj };
   101→  if (field === 'pieceSkin') next.pieceSkin = code;
   102→  else if (field === 'boardSkin') next.boardSkin = code;
   103→  else if (field === 'background') next.background = code;
   104→  else if (field === 'effectSkin') next.effectSkin = code;
   105→  else if (field === 'avatarItem') next.avatarItem = code;
   106→  else next.titleItem = code;
   107→  return next;
   108→}

   112→function ItemPreview({ item, glow }: { item: StoreItemDTO; glow: boolean }) {
   113→  const d = item.data as { cell?: string; cell2?: string; hole?: string; numeral?: string; colors?: string[]; accent?: string };
   114→
   115→  switch (item.category) {
   116→    case 'EMOTE': {
   117→      const pack = (item.data as { emotes?: { id: string; emoji: string }[] }).emotes ?? [];

   120→          {pack.map((e) => (

   123→            </span>

   327→const CONFETTI_COLORS = ['#fbbf24', '#f472b6', '#34d399', '#c084fc', '#f43f5e', '#38bdf8', '#fde047'];

   329→/** Lightweight confetti — absolutely-positioned animated divs, no new deps. */
   330→function Confetti({ seed }: { seed: number }) {
   331→  const parts = useMemo(
   332→    () =>
   333→      Array.from({ length: 30 }, (_, i) => ({
   334→        id: i,
   335→        x: (Math.random() - 0.5) * 340,
   336→        y: -(70 + Math.random() * 170),
   337→        rot: Math.random() * 720 - 360,
   338→        dur: 1 + Math.random(),
   339→        delay: Math.random() * 0.18,
   340→        c: CONFETTI_COLORS[i % CONFETTI_COLORS.length],
   341→        w: 5 + Math.random() * 6,
   342→      })),
   343→    [seed],
   344→  );
   345→  return (
   346→    <div className="pointer-events-none absolute inset-0 overflow-hidden" aria-hidden>
   347→      {parts.map((p) => (
   348→        <motion.span
   349→          key={`${seed}-${p.id}`}
   350→          className="absolute left-1/2 top-[38%] block rounded-[2px]"
   351→          style={{ width: p.w, height: Math.round(p.w * 0.5), background: p.c }}
   352→          initial={{ x: 0, y: 0, opacity: 1, rotate: 0 }}
   353→          animate={{ x: p.x, y: [0, p.y, 330], opacity: [1, 1, 0], rotate: p.rot }}
   354→          transition={{ duration: p.dur + 0.9, delay: p.delay, ease: 'easeOut', times: [0, 0.42, 1] }}
   355→        />
   356→      ))}
   357→    </div>
   358→  );
   359→}

   361→/** Big reward icon inside the reveal ring (reuses the item renderers above). */
   362→function RewardVisual({ kind, itemCode, catalogItem }: { kind: DropKind; itemCode: string | null; catalogItem: StoreItemDTO | null }) {
   363→  if (kind === 'COINS') return <span className="text-5xl drop-shadow-[0_0_16px_rgba(251,191,36,.85)]" aria-hidden>🪙</span>;
   364→  if (kind === 'XP') return <span className="text-5xl text-violet-300 drop-shadow-[0_0_16px_rgba(192,132,252,.85)]" aria-hidden>✦</span>;
   365→  const code = itemCode ?? '';
   366→  const prefix = code.split('_')[0] ?? '';
   367→  if (prefix === 'piece') return <PieceDisc code={code} side={1} size={72} />;
   368→  if (prefix === 'avatar') return <AvatarBubble code={code} size={72} glow />;
   369→  if (catalogItem) return <ItemPreview item={catalogItem} glow />;
   370→  return <span className="text-5xl" aria-hidden>{ITEM_EMOJI[prefix] ?? '📦'}</span>;
   371→}
