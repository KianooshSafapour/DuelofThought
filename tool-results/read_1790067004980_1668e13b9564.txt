     1→'use client';
     2→
     3→/**
     4→ * PlayView — mode selection lobby: vs AI (3 difficulties), local 2P,
     5→ * online (ranked / casual / private room / friend challenges).
     6→ * A lightweight "presence" socket lives in the lobby: it registers the player
     7→ * in the park, receives friend challenges, and is handed off to the game
     8→ * pipeline when a challenge is accepted. Renders GameScreen when a match starts.
     9→ */
    10→import { useEffect, useRef, useState } from 'react';
    11→import { useApp } from '@/store/app';
    12→import { GameScreen, OnlineGameHooks, OnlineRewards } from '@/components/game/GameScreen';
    13→import { FriendsPanel } from '@/components/game/FriendsPanel';
    14→import { GameState } from '@/lib/game/engine';
    15→import { Difficulty } from '@/lib/game/ai';
    16→import { BOARD_CATALOG } from '@/lib/game/boards';
    17→import { Button } from '@/components/ui/button';
    18→import { Dialog, DialogContent, DialogDescription, DialogFooter, DialogHeader, DialogTitle } from '@/components/ui/dialog';
    19→import { Input } from '@/components/ui/input';
    20→import { toast } from 'sonner';
    21→import { sfx } from '@/lib/sound';
    22→import { localizeDigits } from '@/lib/i18n';
    23→import { cn } from '@/lib/utils';
    24→import { AvatarBubble } from '@/components/shared/PixelSprite';
    25→import { Bot, Users, Globe, Crown, Loader2, LogIn, Plus, X, Swords, Copy, Zap } from 'lucide-react';
    26→import { io, Socket } from 'socket.io-client';
    27→
    28→type Phase = 'lobby' | 'game';
    29→
    30→interface IncomingChallenge {
    31→  id: string;
    32→  at: number;
    33→  from: { id: string; username: string; avatarItem: string; rating: number; level: number };
    34→}
    35→
    36→const CHALLENGE_TTL_S = 30;
    37→
    38→export function PlayView() {
    39→  const { t, locale, user, gameSettings, setGameSettings } = useApp();
    40→  const [phase, setPhase] = useState<Phase>('lobby');
    41→  const [onlineHooks, setOnlineHooks] = useState<OnlineGameHooks | null>(null);
    42→  const [onlineState, setOnlineState] = useState<GameState | null>(null);
    43→  const [onlineRewards, setOnlineRewards] = useState<OnlineRewards | null>(null);
    44→  const socketRef = useRef<Socket | null>(null);
    45→  const presenceRef = useRef<Socket | null>(null);
    46→  const [searching, setSearching] = useState(false);
    47→  const [waitingRoom, setWaitingRoom] = useState<string | null>(null);
    48→  const [presence, setPresence] = useState<{ players: number; rooms: number } | null>(null);
    49→  const [joinOpen, setJoinOpen] = useState(false);
    50→  const [joinCode, setJoinCode] = useState('');
    51→  const [joinError, setJoinError] = useState(false);
    52→  const [challengeOut, setChallengeOut] = useState<{ id: string; name: string } | null>(null);
    53→  const [challengeIn, setChallengeIn] = useState<IncomingChallenge | null>(null);
    54→  const [accepting, setAccepting] = useState(false);
    55→  const [leftSec, setLeftSec] = useState(CHALLENGE_TTL_S);
    56→
    57→  const userId = user?.id ?? null;
    58→  const phaseRef = useRef(phase);
    59→  useEffect(() => { phaseRef.current = phase; }, [phase]);
    60→
    61→  /* ---------------- online plumbing ---------------- */
    62→  useEffect(() => {
    63→    return () => {
    64→      socketRef.current?.disconnect();
    65→      presenceRef.current?.disconnect();
    66→    };
    67→  }, []);
    68→
    69→  /* live park presence — polls the duel-socket health endpoint via the gateway */
    70→  useEffect(() => {
    71→    if (phase !== 'lobby') return;
    72→    let live = true;
    73→    const poll = () => {
    74→      fetch('/?XTransformPort=3003', { cache: 'no-store' })
    75→        .then((r) => (r.ok ? r.json() : null))
    76→        .then((d: { players?: number; rooms?: number } | null) => {
    77→          if (live && d && typeof d.players === 'number') setPresence({ players: d.players, rooms: d.rooms ?? 0 });
    78→        })
    79→        .catch(() => undefined);
    80→    };
    81→    poll();
    82→    const id = setInterval(poll, 15_000);
    83→    return () => { live = false; clearInterval(id); };
    84→  }, [phase]);
    85→
    86→  /* ---------------- shared in-game event pipeline ----------------
    87→   * Attached to whichever socket ends up carrying the match (queue socket,
    88→   * private-room socket, or the handed-off lobby presence socket).
    89→   * reassertOnReconnect: game sockets re-emit their queue intent after a
    90→   * transient drop (R7 recovery); the idle lobby socket must NOT — defaulting
    91→   * true there made every lobby browser phantom-queue into casual matchmaking
    92→   * whenever the referee service hiccuped. */
    93→  const attachGameHandlers = (socket: Socket, reassertOnReconnect: boolean) => {
    94→    const chat: { from: string; text: string }[] = [];
    95→    const emotes: { from: string; id: string }[] = [];
    96→    let inLiveGame = reassertOnReconnect; // cleared when the final state (result) arrives
    97→
    98→    socket.on('match', (d: {
    99→      yourSide: 1 | 2; boardId: string; ranked?: boolean;
   100→      opponent: { username: string; avatarItem: string; rating: number; level: number; pieceSkin?: string; boardSkin?: string };
   101→    }) => {
   102→      chat.length = 0;
   103→      emotes.length = 0;
   104→      inLiveGame = true;
   105→      const hooks: OnlineGameHooks & { socket: Socket; boardId: string } = {
   106→        socket,
   107→        boardId: d.boardId,
   108→        mySide: d.yourSide,
   109→        opponentName: d.opponent.username,
   110→        opponentAvatar: d.opponent.avatarItem,
   111→        opponentPieceSkin: d.opponent.pieceSkin,
   112→        chat,
   113→        emotes,
   114→        sendMove: (col: number) => socket.emit('move', { col }),
   115→        sendResign: () => socket.emit('resign'),
   116→        sendRematch: () => socket.emit('rematch:accept'),
   117→        sendChat: (text: string) => socket.emit('chat', { text }),
   118→        sendEmote: (id: string) => socket.emit('emote', { id }),
   119→      };
   120→      setOnlineHooks(hooks);
   121→      setOnlineState(null);
   122→      setOnlineRewards(null);
   123→      setSearching(false);
   124→      setWaitingRoom(null);
   125→      setChallengeOut(null);
   126→      setChallengeIn(null);
   127→      setGameSettings({ boardId: d.boardId });
   128→      setPhase('game');
   129→      sfx.capture();
   130→    });
   131→
   132→    socket.on('move:bad', () => { sfx.bad(); toast.error(t('error')); });
   133→    socket.on('state', (d: { state: GameState }) => {
   134→      const s = typeof d.state === 'string' ? (JSON.parse(d.state) as GameState) : d.state;
   135→      if (s.result) inLiveGame = false;
   136→      setOnlineState(s);
   137→    });
   138→    socket.on('game:rewards', (d: OnlineRewards) => {
   139→      setOnlineRewards(d);
   140→      void useApp.getState().refreshSession();
   141→    });
   142→    socket.on('chat', (d: { from: string; text: string }) => {
   143→      chat.push(d);
   144→      sfx.chat();
   145→      setOnlineHooks((h) => (h ? { ...h, chat: [...chat] } : h));
   146→    });
   147→    socket.on('emote', (d: { from: string; id: string }) => {
   148→      emotes.push(d);
   149→      sfx.chat();
   150→      setOnlineHooks((h) => (h ? { ...h, emotes: [...emotes] } : h));
   151→      setTimeout(() => {
   152→        emotes.splice(0, emotes.length);
   153→        setOnlineHooks((h) => (h ? { ...h, emotes: [] } : h));
   154→      }, 4000);
   155→    });
   156→    socket.on('chat:blocked', () => toast.error(t('chat_blocked')));
   157→    socket.on('opponent:left', () => toast.info(t('opponent_left')));
   158→    // mid-game reconnect: after a transient drop the manager retries, and on
   159→    // success we re-assert the intent — the server resumes an ACTIVE room
   160→    // (rejoin) instead of queueing, so the board continues seamlessly
   161→    socket.io.on('reconnect', () => {
   162→      if (inLiveGame) socket.emit('queue:join', { ranked: false });
   163→    });
   164→    socket.on('game:over', () => { inLiveGame = false; /* server pushes final state via state event */ });
   165→  };
   166→
   167→  /* ---------------- lobby presence socket (friends + challenges) ---------------- */
   168→  useEffect(() => {
   169→    if (phase !== 'lobby' || !userId) return;
   170→    let handedOff = false;
   171→    const socket = io('/?XTransformPort=3003', {
   172→      transports: ['websocket', 'polling'],
   173→      reconnectionAttempts: 5,
   174→    });
   175→    presenceRef.current = socket;
   176→    attachGameHandlers(socket, false); // lobby socket: presence + challenges only
   177→
   178→    socket.on('connect', () => socket.emit('presence:hello'));
   179→    socket.io.on('reconnect', () => socket.emit('presence:hello'));
   180→
   181→    socket.on('challenge:sent', (d: { id: string; to: { username: string } }) => {
   182→      setChallengeOut({ id: d.id, name: d.to.username });
   183→      toast.info(`${t('challenge_sent')} — ${d.to.username}`);
   184→    });
   185→    socket.on('challenge:incoming', (d: IncomingChallenge) => {
   186→      // challenges are only receivable in the park lobby — mid-duel we silently
   187→      // auto-decline so the sender isn't left waiting for the 30s TTL
   188→      if (phaseRef.current !== 'lobby') {
   189→        socket.emit('challenge:decline', { id: d.id });
   190→        return;
   191→      }
   192→      setChallengeIn({ ...d, at: Date.now() });
   193→      sfx.achievement();
   194→    });
   195→    socket.on('challenge:declined', () => {
   196→      setChallengeOut(null);
   197→      sfx.bad();
   198→      toast.info(t('challenge_declined_toast'));
   199→    });
   200→    socket.on('challenge:cancelled', () => {
   201→      setChallengeOut((c) => (c ? null : c));
   202→      toast.info(t('challenge_cancelled_toast'));
   203→    });
   204→    socket.on('challenge:expired', () => {
   205→      setChallengeOut(null);
   206→      setChallengeIn(null);
   207→      toast.info(t('challenge_expired_toast'));
   208→    });
   209→    socket.on('error', (d: { code?: string }) => {
   210→      setChallengeOut(null);
   211→      setAccepting(false);
   212→      const map: Record<string, string> = {
   213→        CHALLENGE_BUSY: t('challenge_busy_toast'),
   214→        CHALLENGE_GONE: t('challenge_gone_toast'),
   215→        CHALLENGE_DUP: t('e_fr_pending'),
   216→        CHALLENGE_SELF: t('e_fr_self'),
   217→        NOT_FOUND: t('challenge_gone_toast'),
   218→      };
   219→      if (d?.code && map[d.code]) toast.error(map[d.code]);
   220→    });
   221→    socket.on('disconnect', () => {
   222→      setChallengeOut((c) => (c ? null : c));
   223→    });
   224→
   225→    // a match can arrive on this socket when a friend challenge is accepted —
   226→    // hand the socket over to the game pipeline instead of tearing it down
   227→    socket.on('match', () => {
   228→      handedOff = true;
   229→      presenceRef.current = null;
   230→      socketRef.current = socket;
   231→      setAccepting(false);
   232→    });
   233→
   234→    return () => {
   235→      presenceRef.current = null;
   236→      if (!handedOff) socket.disconnect();
   237→    };
   238→  }, [phase, userId]);
   239→
   240→  /* cross-view challenge handoff — the leaderboard (or profile) sets
   241→   * pendingChallenge and navigates here; once the lobby socket is live we fire
   242→   * the challenge and clear the handoff (retry ~10s, then give up gracefully) */
   243→  const pendingChallenge = useApp((s) => s.pendingChallenge);
   244→  const setPendingChallenge = useApp((s) => s.setPendingChallenge);
   245→  useEffect(() => {
   246→    if (!pendingChallenge || phase !== 'lobby' || !userId) return;
   247→    let tries = 0;
   248→    const iv = setInterval(() => {
   249→      tries += 1;
   250→      const socket = presenceRef.current;
   251→      if (socket?.connected) {
   252→        clearInterval(iv);
   253→        socket.emit('challenge:send', { toUserId: pendingChallenge.userId });
   254→        setPendingChallenge(null);
   255→      } else if (tries > 25) {
   256→        clearInterval(iv);
   257→        setPendingChallenge(null);
   258→        toast.error(t('challenge_gone_toast'));
   259→      }
   260→    }, 400);
   261→    return () => clearInterval(iv);
   262→  }, [pendingChallenge, phase, userId, setPendingChallenge, t]);
   263→
   264→  /* countdown bar for the incoming challenge dialog — deadline derived from
   265→   * arrival time so no synchronous setState is needed inside the effect */
   266→  useEffect(() => {
   267→    if (!challengeIn) return;
   268→    const endAt = challengeIn.at + CHALLENGE_TTL_S * 1000;
   269→    const iv = setInterval(() => {
   270→      const left = Math.max(0, Math.ceil((endAt - Date.now()) / 1000));
   271→      setLeftSec(left);
   272→      if (left <= 0) { clearInterval(iv); setChallengeIn(null); setAccepting(false); }
   273→    }, 400);
   274→    return () => clearInterval(iv);
   275→  }, [challengeIn]);
   276→
   277→  const dropSocket = () => {
   278→    socketRef.current?.disconnect();
   279→    socketRef.current = null;
   280→  };
   281→
   282→  const cancelQueue = () => {
   283→    socketRef.current?.emit('queue:leave');
   284→    dropSocket();
   285→    setSearching(false);
   286→    setWaitingRoom(null);
   287→    sfx.click();
   288→  };
   289→
   290→  const sendChallenge = (friendId: string) => {
   291→    if (!presenceRef.current) { toast.error(t('error')); return; }
   292→    presenceRef.current.emit('challenge:send', { toUserId: friendId });
   293→  };
   294→
   295→  const cancelChallenge = () => {
   296→    presenceRef.current?.emit('challenge:cancel');
   297→    setChallengeOut(null);
   298→    sfx.click();
   299→  };
   300→
   301→  const acceptChallenge = () => {
   302→    if (!challengeIn || !presenceRef.current) return;
   303→    setAccepting(true);
   304→    presenceRef.current.emit('challenge:accept', { id: challengeIn.id });
   305→  };
   306→
   307→  const declineChallenge = () => {
   308→    if (!challengeIn) return;
   309→    presenceRef.current?.emit('challenge:decline', { id: challengeIn.id });
   310→    setChallengeIn(null);
   311→    setAccepting(false);
   312→    sfx.click();
   313→  };
   314→
   315→  const startOnline = (ranked: boolean, roomCode?: string) => {
   316→    if (!user) { toast.error(t('e_guest_upgrade')); return; }
   317→    if (socketRef.current) socketRef.current.disconnect();
   318→    setJoinOpen(false);
   319→    setSearching(roomCode === undefined);
   320→    setWaitingRoom(null);
   321→    // Gateway routing: the port MUST ride in the URL query (XTransformPort).
   322→    // The httpOnly session cookie is forwarded automatically (same-origin).
   323→    const socket = io('/?XTransformPort=3003', {
   324→      transports: ['websocket', 'polling'],
   325→      reconnectionAttempts: 10,
   326→      reconnectionDelay: 1500,
   327→    });
   328→    socketRef.current = socket;
   329→    attachGameHandlers(socket, true); // game socket: re-assert queue intent on reconnect
   330→
   331→    socket.on('connect', () => {
   332→      if (roomCode) socket.emit('room:join', { code: roomCode });
   333→      else if (roomCode === '') socket.emit('room:create');
   334→      else socket.emit('queue:join', { ranked });
   335→    });
   336→    // transient drops: re-assert the intent after reconnect (server dedups /
   337→    // resumes an active room, so this is idempotent)
   338→    socket.io.on('reconnect', () => {
   339→      if (roomCode) socket.emit('room:join', { code: roomCode });
   340→      else if (roomCode === '') socket.emit('room:create');
   341→      else socket.emit('queue:join', { ranked });
   342→    });
   343→    socket.on('connect_error', () => { setSearching(false); setWaitingRoom(null); toast.error(t('connection_lost')); });
   344→    socket.on('disconnect', () => { setSearching(false); setWaitingRoom(null); });
   345→
   346→    socket.on('error', (d: { code?: string }) => {
   347→      setSearching(false);
   348→      setWaitingRoom(null);
   349→      toast.error(d?.code === 'BANNED' ? t('e_banned') : d?.code === 'ROOM_NOT_FOUND' ? t('invalid_code') : t('error'));
   350→    });
   351→
   352→    socket.on('queued', () => { setSearching(true); toast.info(t('finding_opponent')); });
   353→    socket.on('room:created', (d: { code: string }) => {
   354→      navigator.clipboard?.writeText(d.code).catch(() => undefined);
   355→      setWaitingRoom(d.code);
   356→    });
   357→  };
   358→
   359→  /* local / ai start — socket cleanup handled by onExit / cancelQueue (ref reads stay out of render) */
   360→  const startLocal = (mode: 'ai' | 'local') => {
   361→    sfx.click();
   362→    setOnlineHooks(null);
   363→    setPhase('game');
   364→  };
   365→
   366→  const submitJoin = () => {
   367→    const code = joinCode.trim().toUpperCase();
   368→    if (!/^[A-Z0-9]{4,6}$/.test(code)) { setJoinError(true); return; }
   369→    setJoinError(false);
   370→    startOnline(false, code);
   371→  };
   372→
   373→  if (phase === 'game') {
   374→    return (
   375→      <GameScreen
   376→        mode={onlineHooks ? 'online' : gameSettings.mode === 'local' ? 'local' : 'ai'}
   377→        aiLevel={gameSettings.aiLevel}
   378→        boardId={gameSettings.boardId}
   379→        online={onlineHooks ?? undefined}
   380→        onlineState={onlineHooks ? onlineState : null}
   381→        onlineRewards={onlineHooks ? onlineRewards : null}
   382→        onExit={() => {
   383→          socketRef.current?.disconnect();
   384→          socketRef.current = null;
   385→          setOnlineHooks(null);
   386→          setOnlineState(null);
   387→          setOnlineRewards(null);
   388→          setPhase('lobby');
   389→        }}
   390→      />
   391→    );
   392→  }
   393→
   394→  const cards: {
   395→    key: string; title: string; desc: string; icon: React.ReactNode;
   396→    onClick: () => void; disabled?: boolean; accent: string;
   397→  }[] = [
   398→    { key: 'easy', title: `${t('mode_ai')} — ${t('ai_easy')}`, desc: '★☆☆', icon: <Bot className="h-6 w-6" />, accent: 'from-emerald-600/30', onClick: () => { setGameSettings({ mode: 'ai', aiLevel: 'EASY' }); startLocal('ai'); } },
   399→    { key: 'medium', title: `${t('mode_ai')} — ${t('ai_medium')}`, desc: '★★☆', icon: <Bot className="h-6 w-6" />, accent: 'from-amber-600/30', onClick: () => { setGameSettings({ mode: 'ai', aiLevel: 'MEDIUM' }); startLocal('ai'); } },
   400→    { key: 'hard', title: `${t('mode_ai')} — ${t('ai_hard')}`, desc: '★★★', icon: <Crown className="h-6 w-6" />, accent: 'from-rose-600/30', onClick: () => { setGameSettings({ mode: 'ai', aiLevel: 'HARD' }); startLocal('ai'); } },
   401→    { key: 'local', title: t('mode_local'), desc: t('duel_desc').slice(0, 40), icon: <Users className="h-6 w-6" />, accent: 'from-fuchsia-600/30', onClick: () => { setGameSettings({ mode: 'local' }); startLocal('local'); } },
   402→  ];
   403→
   404→  const canChallenge = !searching && !waitingRoom && !challengeOut && !challengeIn;
   405→
   406→  return (
   407→    <div className="mx-auto w-full max-w-4xl px-4 pb-8">
   408→      <h1 className="mb-1 text-center text-2xl font-black"><span className="gold-text">{t('game_new')}</span></h1>
   409→      <p className="mb-5 text-center text-xs text-muted-foreground">{t('tagline')}</p>
   410→
   411→      {/* board selection */}
   412→      <div className="royal-card mb-5 rounded-2xl p-4">
   413→        <div className="mb-2 text-sm font-bold">{t('choose_board')}</div>
   414→        <div className="grid grid-cols-2 gap-2 sm:grid-cols-4">
   415→          {BOARD_CATALOG.map((b) => (
   416→            <button
   417→              key={b.id}
   418→              disabled={searching}
   419→              onClick={() => { setGameSettings({ boardId: b.id }); sfx.click(); }}
   420→              className={`rounded-xl border p-2.5 text-center transition-all disabled:opacity-40 ${
   421→                gameSettings.boardId === b.id ? 'border-amber-400 bg-amber-400/10 glow-gold' : 'border-border hover:border-amber-400/50'
   422→              }`}
   423→            >
   424→              <MiniBoardPreview id={b.id} />
   425→              <div className="mt-1 text-xs font-bold">
   426→                {t(`board_${b.id}` as 'board_classic' | 'board_daily' | 'board_symmetry' | 'board_storm')}
   427→              </div>
   428→              {b.id === 'daily' && <div className="text-[9px] text-muted-foreground">{localizeDigits(new Date().toISOString().slice(5, 10).replace('-', '/'), locale)}</div>}
   429→            </button>
   430→          ))}
   431→        </div>
   432→        {gameSettings.boardId === 'daily' && <div className="mt-2 text-center text-[10px] text-muted-foreground">{t('daily_board_note')}</div>}
   433→      </div>
   434→
   435→      {/* mode cards */}
   436→      <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
   437→        {cards.map((c) => (
   438→          <button
   439→            key={c.key}
   440→            disabled={searching}
   441→            onClick={c.onClick}
   442→            className={`royal-card group rounded-2xl bg-gradient-to-br ${c.accent} to-transparent p-4 text-start transition-all hover:scale-[1.02] hover:glow-purple disabled:pointer-events-none disabled:opacity-40`}
   443→          >
   444→            <div className="flex items-center gap-3">
   445→              <div className="rounded-xl bg-black/30 p-2.5 text-amber-400">{c.icon}</div>
   446→              <div>
   447→                <div className="font-bold">{c.title}</div>
   448→                <div className="text-xs text-muted-foreground">{c.desc}</div>
   449→              </div>
   450→            </div>
   451→          </button>
   452→        ))}
   453→      </div>
   454→
   455→      {/* online section */}
   456→      <div className="royal-card mt-4 rounded-2xl p-4">
   457→        <div className="mb-3 flex items-center gap-2 font-bold"><Globe className="h-5 w-5 text-emerald-400" /> {t('online')}</div>
   458→        <div className="grid grid-cols-1 gap-2 sm:grid-cols-3">
   459→          <Button className="glow-gold" onClick={() => startOnline(true)} disabled={!user || searching}>
   460→            <Crown className="h-4 w-4" /> {t('ranked')}
   461→          </Button>
   462→          <Button variant="secondary" onClick={() => startOnline(false)} disabled={!user || searching}>
   463→            <Globe className="h-4 w-4" /> {t('casual')}
   464→          </Button>
   465→          <Button
   466→            variant="outline"
   467→            disabled={!user || searching}
   468→            onClick={() => { setJoinCode(''); setJoinError(false); setJoinOpen(true); sfx.click(); }}
   469→          >
   470→            <LogIn className="h-4 w-4" /> {t('join_room')}
   471→          </Button>
   472→        </div>
   473→        <div className="mt-2">
   474→          <Button
   475→            variant="ghost"
   476→            size="sm"
   477→            disabled={!user || searching || !!waitingRoom}
   478→            onClick={() => startOnline(false, '')}
   479→          >
   480→            <Plus className="h-4 w-4" /> {t('create_room')}
   481→          </Button>
   482→        </div>
   483→        {!user && <div className="mt-1 text-center text-xs text-rose-400">{t('e_guest_upgrade')}</div>}
   484→
   485→        {searching || waitingRoom ? (
   486→          <div className="mt-3 flex flex-col items-center gap-2 rounded-xl border border-amber-400/30 bg-gradient-to-b from-amber-400/10 to-transparent p-4">
   487→            <div className="relative h-16 w-16" aria-hidden>
   488→              <span className="absolute inset-0 rounded-full border-2 border-amber-400/50 animate-ping" />
   489→              <span className="absolute inset-0 rounded-full border-2 border-amber-400/35 animate-ping" style={{ animationDelay: '0.55s' }} />
   490→              <span className="absolute inset-0 rounded-full border-2 border-amber-400/20 animate-ping" style={{ animationDelay: '1.1s' }} />
   491→              <div className="absolute inset-2 flex items-center justify-center rounded-full bg-gradient-to-br from-amber-400/40 to-amber-700/30 text-amber-300">
   492→                <Swords className="h-5 w-5" />
   493→              </div>
   494→            </div>
   495→            {waitingRoom && (
   496→              <button
   497→                className="group flex items-center gap-2 rounded-lg border border-amber-400/40 bg-black/30 px-4 py-1.5 transition-colors hover:border-amber-400"
   498→                onClick={() => { navigator.clipboard?.writeText(waitingRoom).catch(() => undefined); sfx.click(); toast.success(t('copied')); }}
   499→                title={t('copied')}
   500→              >
   501→                <span className="font-mono text-xl font-black tracking-[0.3em] text-amber-300" dir="ltr">{waitingRoom}</span>
   502→                <Copy className="h-4 w-4 text-muted-foreground group-hover:text-amber-300" />
   503→              </button>
   504→            )}
   505→            <div className="text-xs font-black text-amber-300">{waitingRoom ? t('waiting_opponent') : t('finding_opponent')}</div>
   506→            <Button variant="ghost" size="sm" className="h-7 text-muted-foreground hover:text-foreground" onClick={cancelQueue}>
   507→              <X className="h-3.5 w-3.5" /> {t('cancel_queue')}
   508→            </Button>
   509→          </div>
   510→        ) : presence ? (
   511→          <div className="mt-2 flex items-center justify-center gap-2 text-[11px] text-muted-foreground" title={t('online')}>
   512→            <span className="relative flex h-2 w-2" aria-hidden>
   513→              <span className="absolute inline-flex h-full w-full animate-ping rounded-full bg-emerald-400 opacity-75" />
   514→              <span className="relative inline-flex h-2 w-2 rounded-full bg-emerald-500" />
   515→            </span>
   516→            <span className="inline-flex items-center gap-1">
   517→              <Users className="h-3 w-3" /> {localizeDigits(presence.players, locale)} {t('players_online')}
   518→            </span>
   519→            <span aria-hidden>·</span>
   520→            <span className="inline-flex items-center gap-1">
   521→              <Swords className="h-3 w-3" /> {localizeDigits(presence.rooms, locale)} {t('live_duels')}
   522→            </span>
   523→          </div>
   524→        ) : (
   525→          <div className="mt-2 flex items-center justify-center gap-1 text-[10px] text-muted-foreground">
   526→            <Loader2 className="h-3 w-3 animate-spin" /> {t('loading')}
   527→          </div>
   528→        )}
   529→      </div>
   530→
   531→      {/* friends */}
   532→      <FriendsPanel
   533→        canChallenge={canChallenge}
   534→        challengeOutName={challengeOut?.name ?? null}
   535→        onChallenge={sendChallenge}
   536→        onCancelChallenge={cancelChallenge}
   537→      />
   538→
   539→      {/* join-by-code dialog */}
   540→      <Dialog open={joinOpen} onOpenChange={setJoinOpen}>
   541→        <DialogContent className="sm:max-w-xs" aria-describedby={undefined}>
   542→          <DialogHeader>
   543→            <DialogTitle className="flex items-center gap-2">
   544→              <LogIn className="h-5 w-5 text-amber-400" /> {t('join_room')}
   545→            </DialogTitle>
   546→            <DialogDescription>{t('join_code_hint')}</DialogDescription>
   547→          </DialogHeader>
   548→          <Input
   549→            autoFocus
   550→            value={joinCode}
   551→            onChange={(e) => { setJoinCode(e.target.value.toUpperCase().slice(0, 6)); setJoinError(false); }}
   552→            onKeyDown={(e) => { if (e.key === 'Enter') submitJoin(); }}
   553→            placeholder="XXXX"
   554→            maxLength={6}
   555→            dir="ltr"
   556→            className={cn(
   557→              'text-center font-mono text-xl font-black tracking-[0.4em]',
   558→              joinError ? 'border-rose-500 focus-visible:ring-rose-500' : 'border-amber-400/50 focus-visible:ring-amber-400',
   559→            )}
   560→            aria-invalid={joinError}
   561→          />
   562→          {joinError && <p className="text-center text-xs font-bold text-rose-400">{t('invalid_code')}</p>}
   563→          <DialogFooter>
   564→            <Button variant="ghost" onClick={() => setJoinOpen(false)}>{t('cancel')}</Button>
   565→            <Button className="glow-gold" onClick={submitJoin} disabled={!/^[A-Z0-9]{4,6}$/.test(joinCode.trim())}>
   566→              <LogIn className="h-4 w-4" /> {t('join_action')}
   567→            </Button>
   568→          </DialogFooter>
   569→        </DialogContent>
   570→      </Dialog>
   571→
   572→      {/* incoming friend-challenge dialog */}
   573→      <Dialog open={!!challengeIn} onOpenChange={(o) => { if (!o && !accepting) declineChallenge(); }}>
   574→        <DialogContent className="royal-card sm:max-w-sm" aria-describedby={undefined}>
   575→          <DialogHeader>
   576→            <DialogTitle className="flex items-center justify-center gap-2 text-center gold-text">
   577→              <Zap className="h-5 w-5 text-amber-400" /> {t('challenge_incoming')}
   578→            </DialogTitle>
   579→            <DialogDescription className="text-center">{t('challenge_incoming_sub')}</DialogDescription>
   580→          </DialogHeader>
   581→          {challengeIn && (
   582→            <div className="flex flex-col items-center gap-3 py-1">
   583→              <div className="relative">
   584→                <span className="absolute -inset-3 rounded-full border-2 border-amber-400/50 animate-ping" aria-hidden />
   585→                <span className="absolute -inset-3 rounded-full border-2 border-amber-400/25 animate-ping" style={{ animationDelay: '0.6s' }} aria-hidden />
   586→                <AvatarBubble code={challengeIn.from.avatarItem} size={64} glow />
   587→              </div>
   588→              <div className="text-center">
   589→                <div className="text-lg font-black" dir="auto">{challengeIn.from.username}</div>
   590→                <div className="mt-0.5 flex items-center justify-center gap-2 text-[11px] text-muted-foreground">
   591→                  <span>⭐ {t('level')} {localizeDigits(challengeIn.from.level, locale)}</span>
   592→                  <span aria-hidden>·</span>
   593→                  <span>🏆 {localizeDigits(challengeIn.from.rating, locale)}</span>
   594→                </div>
   595→              </div>
   596→              {/* countdown bar */}
   597→              <div className="h-1.5 w-full overflow-hidden rounded-full bg-muted" role="timer" aria-label={t('challenge_wait')}>
   598→                <div
   599→                  className={cn('h-full rounded-full transition-all duration-1000 ease-linear', leftSec > 8 ? 'bg-amber-400' : 'bg-rose-500')}
   600→                  style={{ width: `${Math.max(0, (leftSec / CHALLENGE_TTL_S) * 100)}%` }}
   601→                />
   602→              </div>
   603→            </div>
   604→          )}
   605→          <DialogFooter className="gap-2">
   606→            <Button variant="ghost" onClick={declineChallenge} disabled={accepting}>{t('challenge_no')}</Button>
   607→            <Button className="glow-gold" onClick={acceptChallenge} disabled={accepting}>
   608→              {accepting ? <Loader2 className="h-4 w-4 animate-spin" /> : <Swords className="h-4 w-4" />}
   609→              {t('challenge_accept')}
   610→            </Button>
   611→          </DialogFooter>
   612→        </DialogContent>
   613→      </Dialog>
   614→    </div>
   615→  );
   616→}

   618→function MiniBoardPreview({ id }: { id: string }) {
   619→  const cols = 9, rows = 7;
   620→  // deterministic mini-preview: hash id to cells
   621→  let seed = 0;
   622→  for (let i = 0; i < id.length; i++) seed = (seed * 31 + id.charCodeAt(i)) | 0;
   623→  const rnd = (n: number) => { const x = Math.sin(seed + n * 7919) * 10000; return x - Math.floor(x); };
   624→  return (
   625→    <div className="mx-auto grid w-full max-w-[90px] gap-[1px] rounded-md bg-amber-900/60 p-[3px]" style={{ gridTemplateColumns: `repeat(${cols},1fr)` }} aria-hidden>
   626→      {Array.from({ length: cols * rows }, (_, i) => (
   627→        <div key={i} className="aspect-square rounded-[1px]" style={{ background: rnd(i) > 0.14 ? '#e3b878' : 'transparent' }} />
   628→      ))}
   629→    </div>
   630→  );
   631→}
