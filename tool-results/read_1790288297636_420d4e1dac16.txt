   866→/** Convert Latin digits to locale digits (fa/ar use Eastern Arabic numerals). */
   867→export function localizeDigits(input: string | number, locale: Locale): string {
   868→  const s = String(input);
   869→  if (locale === 'fa') {
   870→    const faD = ['۰', '۱', '۲', '۳', '۴', '۵', '۶', '۷', '۸', '۹'];
   871→    return s.replace(/[0-9]/g, (d) => faD[+d]);
   872→  }
   873→  if (locale === 'ar') {
   874→    const arD = ['٠', '١', '٢', '٣', '٤', '٥', '٦', '٧', '٨', '٩'];
   875→    return s.replace(/[0-9]/g, (d) => arD[+d]);
   876→  }
   877→  return s;
   878→}
