     1→/**
     2→ * Game content catalog: store items (skins/avatars/backgrounds/titles/effects),
     3→ * achievements, daily quests, rarity tiers. Item `data` JSON drives the
     4→ * procedural pixel-art renderer (client) so no binary art assets are needed.
     5→ */
     6→
     7→export type Rarity = 'COMMON' | 'RARE' | 'EPIC' | 'LEGENDARY' | 'ULTIMATE';
     8→export type ItemCategory = 'PIECE' | 'BOARD' | 'AVATAR' | 'BACKGROUND' | 'EFFECT' | 'TITLE' | 'EMOTE' | 'CELEBRATION';
     9→
    10→export interface ItemDef {
    11→  code: string;
    12→  category: ItemCategory;
    13→  rarity: Rarity;
    14→  price: number;
    15→  requiredLevel: number;
    16→  data: Record<string, unknown>;
    17→}

    26→
    27→/** Every piece skin defines palettes for BOTH sides + a pixel pattern seed. */
    28→const piece = (code: string, rarity: Rarity, price: number, p1: string[], p2: string[], seed: number, nameGlow?: string): ItemDef =>
    29→  ({ code, category: 'PIECE', rarity, price, requiredLevel: rarity === 'ULTIMATE' ? 12 : rarity === 'LEGENDARY' ? 6 : rarity === 'EPIC' ? 3 : 1, data: { p1, p2, seed, glow: nameGlow ?? p1[0] } });
    30→
    31→const avatar = (code: string, rarity: Rarity, price: number, seed: number, pal: string[]): ItemDef =>
    32→  ({ code, category: 'AVATAR', rarity, price, requiredLevel: 1, data: { seed, pal } });
    33→
    34→const bg = (code: string, rarity: Rarity, price: number, kind: string, colors: string[], accent: string): ItemDef =>
    35→  ({ code, category: 'BACKGROUND', rarity, price, requiredLevel: 1, data: { kind, colors, accent } });
    36→
    37→const board = (code: string, rarity: Rarity, price: number, pal: Record<string, string>): ItemDef =>
    38→  ({ code, category: 'BOARD', rarity, price, requiredLevel: rarity === 'ULTIMATE' ? 12 : rarity === 'EPIC' ? 3 : 1, data: pal });
    39→
    40→const fx = (code: string, rarity: Rarity, price: number, colors: string[], shape: string): ItemDef =>
    41→  ({ code, category: 'EFFECT', rarity, price, requiredLevel: 1, data: { colors, shape } });
    42→
    43→const title = (code: string, rarity: Rarity, price: number, reqLevel: number): ItemDef =>
    44→  ({ code, category: 'TITLE', rarity, price, requiredLevel: reqLevel, data: {} });

   366→/** Deterministically pick 3 quests per user per day (seeded by dateKey+userId). */
   367→export function questsForDay(dateKey: string, userId: string): QuestDef[] {
   368→  let seed = 2166136261;
   369→  const str = dateKey + '|' + userId;
   370→  for (let i = 0; i < str.length; i++) {
   371→    seed ^= str.charCodeAt(i);
   372→    seed = Math.imul(seed, 16777619);
   373→  }
   374→  const rnd = (() => { let a = seed >>> 0; return () => { a |= 0; a = (a + 0x6d2b79f5) | 0; let t = Math.imul(a ^ (a >>> 15), 1 | a); t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t; return ((t ^ (t >>> 14)) >>> 0) / 4294967296; }; })();
   375→  const pool = [...QUEST_POOL];
   376→  const picked: QuestDef[] = [];
   377→  while (picked.length < 3 && pool.length) {
   378→    picked.push(pool.splice(Math.floor(rnd() * pool.length), 1)[0]);
   379→  }
   380→  return picked;
   381→}

   383→/* ================= ECONOMY ================= */
   384→export function xpForLevel(level: number): number {
   385→  // XP needed to advance FROM `level` to level+1
   386→  return 100 + (level - 1) * 60;
   387→}
   388→
   389→export function levelFromXp(xp: number): { level: number; into: number; need: number } {
   390→  let level = 1;
   391→  let remaining = xp;
   392→  while (remaining >= xpForLevel(level) && level < 99) {
   393→    remaining -= xpForLevel(level);
   394→    level++;
   395→  }
   396→  return { level, into: remaining, need: xpForLevel(level) };
   397→}

   399→export interface RewardInput {
   400→  mode: 'PVE' | 'LOCAL_PVP' | 'ONLINE';
   401→  outcome: 'WIN' | 'LOSS' | 'DRAW';
   402→  aiLevel?: 'EASY' | 'MEDIUM' | 'HARD';
   403→  ranked?: boolean;
   404→  moves: number;
   405→  streak: number; // current streak BEFORE this game (for win bonus)
   406→}
   407→
   408→export function computeRewards(r: RewardInput): { coins: number; xp: number } {
   409→  let coins = 0;
   410→  let xp = 0;
   411→  if (r.mode === 'ONLINE') {
   412→    if (r.outcome === 'WIN') { coins = 60 + Math.min(r.streak, 5) * 6; xp = 80; }
   413→    else if (r.outcome === 'DRAW') { coins = 28; xp = 40; }
   414→    else { coins = 12; xp = 20; }
   415→  } else if (r.mode === 'PVE') {
   416→    const base = r.aiLevel === 'HARD' ? 45 : r.aiLevel === 'MEDIUM' ? 22 : 8;
   417→    if (r.outcome === 'WIN') { coins = base; xp = 30 + (r.aiLevel === 'HARD' ? 30 : r.aiLevel === 'MEDIUM' ? 12 : 0); }
   418→    else if (r.outcome === 'DRAW') { coins = Math.ceil(base / 3); xp = 15; }
   419→    else { coins = Math.ceil(base / 5); xp = 8; }
   420→  } else { // LOCAL_PVP — fun + XP only (anti-farm: near-zero coins)
   421→    coins = 4; xp = 25;
   422→  }
   423→  return { coins, xp };
   424→}

   426→/** ELO with K=32; returns deltas from p1 perspective */
   427→export function eloDeltas(r1: number, r2: number, p1Score: 0 | 0.5 | 1, k = 32): [number, number] {
   428→  const exp = 1 / (1 + Math.pow(10, (r2 - r1) / 400));
   429→  const d = Math.round(k * (p1Score - exp));
   430→  return [d, -d];
   431→}

   519→/** Weighted random pick (server-side). Returns the winning drop. */
   520→export function rollDrop(drops: DropDef[], rnd: () => number = Math.random): DropDef {
   521→  const total = drops.reduce((s, d) => s + d.weight, 0);
   522→  let r = rnd() * total;
   523→  for (const d of drops) {
   524→    r -= d.weight;
   525→    if (r <= 0) return d;
   526→  }
   527→  return drops[drops.length - 1];
   528→}

   537→/**
   538→ * Score -> rewards. Score scales differ per game; each formula is tuned so a
   539→ * solid mid-tier run ≈ a duel win. Server clamps by hourly cap.
   540→ */
   541→export function arcadeRewards(game: ArcadeGame, score: number): { coins: number; xp: number } {
   542→  if (!Number.isFinite(score) || score <= 0) return { coins: 3, xp: 6 };
   543→  const s = Math.floor(score);
   544→  if (game === 'orbit') {
   545→    // perfect mission ≈ 1000; typical early mission ≈ 300
   546→    const coins = Math.min(90, 12 + Math.floor(s / 12));
   547→    const xp = Math.min(120, 20 + Math.floor(s / 9));
   548→    return { coins, xp };
   549→  }
   550→  if (game === 'synapse') {
   551→    // session score ≈ trials × accuracy × n
   552→    const coins = Math.min(80, 10 + Math.floor(s / 14));
   553→    const xp = Math.min(110, 18 + Math.floor(s / 10));
   554→    return { coins, xp };
   555→  }
   556→  // stellar: mass units forged; a full supernova run ≈ 2000+
   557→  const coins = Math.min(85, 10 + Math.floor(s / 24));
   558→  const xp = Math.min(115, 16 + Math.floor(s / 18));
   559→  return { coins, xp };
   560→}

   562→/** v0.2 reward curves for the three new arcade games (mind-tank wave scores,
   563→ *  boulder yard level scores). Same tuning goal: solid run ≈ a duel win. */
   564→export function arcadeRewardsV2(game: 'mazemind' | 'boulderyard' | 'boulderyard2', score: number): { coins: number; xp: number } {
   565→  if (!Number.isFinite(score) || score <= 0) return { coins: 3, xp: 6 };
   566→  const s = Math.floor(score);
   567→  if (game === 'mazemind') {
   568→    // a full 10-robot run ≈ 3200; casual early death ≈ 500
   569→    const coins = Math.min(100, 12 + Math.floor(s / 26));
   570→    const xp = Math.min(130, 16 + Math.floor(s / 20));
   571→    return { coins, xp };
   572→  }
   573→  if (game === 'boulderyard') {
   574→    // solving ~6 levels ≈ 1400
   575→    const coins = Math.min(90, 10 + Math.floor(s / 18));
   576→    const xp = Math.min(110, 14 + Math.floor(s / 15));
   577→    return { coins, xp };
   578→  }
   579→  // boulderyard2: trickier mechanics, slightly richer
   580→  const coins = Math.min(95, 10 + Math.floor(s / 17));
   581→  const xp = Math.min(120, 14 + Math.floor(s / 14));
   582→  return { coins, xp };
   583→}
