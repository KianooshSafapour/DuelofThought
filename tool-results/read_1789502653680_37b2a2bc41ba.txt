   600→/** Convert Latin digits to locale digits (fa/ar use Eastern Arabic numerals). */
   601→export function localizeDigits(input: string | number, locale: Locale): string {
   602→  const s = String(input);
   603→  if (locale === 'fa') {
   604→    const faD = ['۰', '۱', '۲', '۳', '۴', '۵', '۶', '۷', '۸', '۹'];
   605→    return s.replace(/[0-9]/g, (d) => faD[+d]);
   606→  }
   607→  if (locale === 'ar') {
   608→    const arD = ['٠', '١', '٢', '٣', '٤', '٥', '٦', '٧', '٨', '٩'];
   609→    return s.replace(/[0-9]/g, (d) => arD[+d]);
   610→  }
   611→  return s;
   612→}
