    69→    69→function sumStats(list: ChannelStats[]): ChannelStats {
    70→    70→  return list.reduce(
    71→    71→    (a, c) => ({ hits: a.hits + c.hits, misses: a.misses + c.misses, fa: a.fa + c.fa, opps: a.opps + c.opps }),

    79→    79→
    80→    80→let audioCtx: AudioContext | null = null;
    81→    81→
    82→    82→function ensureAudio(): AudioContext | null {
    83→    83→  if (typeof window === 'undefined') return null;
    84→    84→  if (!audioCtx) {
    85→    85→    try {
    86→    86→      audioCtx = new (window.AudioContext || (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext)();
    87→    87→    } catch {
    88→    88→      return null;
    89→    89→    }
    90→    90→  }
    91→    91→  if (audioCtx.state === 'suspended') void audioCtx.resume();
    92→    92→  return audioCtx;
    93→    93→}
    94→    94→
    95→    95→function releaseAudio() {
    96→    96→  if (audioCtx) {
    97→    97→    const c = audioCtx;
    98→    98→    audioCtx = null;
    99→    99→    void c.close().catch(() => undefined);
   100→   100→  }
   101→   101→}

   103→   103→/** Letter tone: sine fundamental + soft octave harmonic, 350 ms gentle envelope, master gain 0.14. */
   104→   104→function playLetter(idx: number, on: boolean) {
   105→   105→  if (!on) return;
   106→   106→  const ctx = ensureAudio();
   107→   107→  if (!ctx) return;
   108→   108→  const f = SYNAPSE_TONES[idx];
   109→   109→  const t0 = ctx.currentTime;
   110→   110→  const master = ctx.createGain();
   111→   111→  master.gain.value = 0.14;
   112→   112→  master.connect(ctx.destination);
   113→   113→  const voice = (freq: number, vol: number) => {
   114→   114→    const osc = ctx.createOscillator();
   115→   115→    osc.type = 'sine';
   116→   116→    osc.frequency.value = freq;
   117→   117→    const g = ctx.createGain();
   118→   118→    g.gain.setValueAtTime(0, t0);
   119→   119→    g.gain.linearRampToValueAtTime(vol, t0 + 0.02);
   120→   120→    g.gain.exponentialRampToValueAtTime(0.0001, t0 + 0.35);
   121→   121→    osc.connect(g).connect(master);
   122→   122→    osc.start(t0);
   123→   123→    osc.stop(t0 + 0.4);
   124→   124→  };
   125→   125→  voice(f, 1);
   126→   126→  voice(f * 2, 0.25);
   127→   127→}
   128→   128→
   129→   129→/** Tiny press-feedback blips (correct = bright tick, false alarm = low thud). */
   130→   130→function playBlip(kind: 'good' | 'bad', on: boolean) {
   131→   131→  if (!on) return;
   132→   132→  const ctx = ensureAudio();
   133→   133→  if (!ctx) return;
   134→   134→  const t0 = ctx.currentTime;
   135→   135→  const osc = ctx.createOscillator();
   136→   136→  const g = ctx.createGain();
   137→   137→  osc.type = 'sine';
   138→   138→  osc.frequency.value = kind === 'good' ? 880 : 185;
   139→   139→  g.gain.setValueAtTime(0, t0);
   140→   140→  g.gain.linearRampToValueAtTime(kind === 'good' ? 0.07 : 0.06, t0 + 0.01);
   141→   141→  g.gain.exponentialRampToValueAtTime(0.0001, t0 + (kind === 'good' ? 0.09 : 0.16));
   142→   142→  osc.connect(g).connect(ctx.destination);
   143→   143→  osc.start(t0);
   144→   144→  osc.stop(t0 + 0.2);
   145→   145→}

   205→   205→export function SynapseGame({ onExit }: { onExit: () => void }) {
   206→   206→  const { t, locale, user } = useApp();
   207→   207→  const { finish, rewards, open, close } = useArcadeFinish('synapse');
   208→   208→  const isRtl = locale === 'fa' || locale === 'ar';
   209→   209→  const loc = useCallback((v: string | number) => localizeDigits(v, locale), [locale]);
   210→   210→  const pctSign = locale === 'fa' ? '٪' : '%';
   211→   211→
   212→   212→  /* ---------- persistent/local UI state ---------- */
   213→   213→  const [maxDay, setMaxDay] = useState(0);
   214→   214→  const [phase, setPhase] = useState<Phase>('home');
   215→   215→  const [brief, setBrief] = useState<BriefItem | null>(null);
   216→   216→  const [showGlyph, setShowGlyph] = useState(true);
   217→   217→  const [count, setCount] = useState(3);
   218→   218→  const [session, setSession] = useState<{ mode: 'campaign' | 'practice'; day: number; title: string } | null>(null);
   219→   219→
   220→   220→  /* ---------- live HUD state (updated once per trial, never per tick) ---------- */
   221→   221→  const [activeCell, setActiveCell] = useState<number | null>(null);
   222→   222→  const [glyph, setGlyph] = useState<string | null>(null);
   223→   223→  const [trialIdx, setTrialIdx] = useState(-1);
   224→   224→  const [nBack, setNBack] = useState(2);
   225→   225→  const [blockNum, setBlockNum] = useState(1);
   226→   226→  const [score, setScore] = useState(0);
   227→   227→  const [fx, setFxState] = useState<{ pos: Fx; snd: Fx }>({ pos: null, snd: null });
   228→   228→  const [interlude, setInterlude] = useState<{ block: number; prevN: number; nextNV: number; acc: number; hits: number; fas: number } | null>(null);
   229→   229→  const [summary, setSummary] = useState<Summary | null>(null);
   230→   230→
   231→   231→  /* ---------- engine refs (never touched during render) ---------- */
   232→   232→  const timersRef = useRef<number[]>([]);
   233→   233→  const phaseRef = useRef<Phase>('home');
   234→   234→  const nRef = useRef(2);
   235→   235→  const trialRef = useRef(-1);
   236→   236→  const blockNumRef = useRef(1);
   237→   237→  const seqRef = useRef<{ pos: number[]; snd: number[] }>({ pos: [], snd: [] });
   238→   238→  const respRef = useRef<{ pos: boolean; snd: boolean }>({ pos: false, snd: false });
   239→   239→  const statsRef = useRef<{ pos: ChannelStats; snd: ChannelStats }>({ pos: emptyStats(), snd: emptyStats() });
   240→   240→  const sessionRef = useRef<{ mode: 'campaign' | 'practice'; day: number; title: string; startN: number; points: number; blocks: BlockSummary[] }>({
   241→   241→    mode: 'practice', day: 0, title: '', startN: 2, points: 0, blocks: [],
   242→   242→  });
   243→   243→  const soundRef = useRef(true);
   244→   244→  const pressRef = useRef<(ch: Channel) => void>(() => undefined);
   245→   245→
   246→   246→  const go = useCallback((p: Phase) => {
   247→   247→    phaseRef.current = p;
   248→   248→    setPhase(p);
   249→   249→  }, []);
   250→   250→
   251→   251→  /* ---------- timer plumbing ---------- */
   252→   252→  const clearTimers = useCallback(() => {
   253→   253→    for (const id of timersRef.current) window.clearTimeout(id);
   254→   254→    timersRef.current = [];
   255→   255→  }, []);
   256→   256→
   257→   257→  const later = (fn: () => void, ms: number) => {
   258→   258→    const id = window.setTimeout(fn, ms);
   259→   259→    timersRef.current.push(id);
   260→   260→  };
   261→   261→
   262→   262→  const setFx = useCallback((ch: Channel, kind: Fx) => {
   263→   263→    setFxState((prev) => ({ ...prev, [ch]: kind }));
   264→   264→    const id = window.setTimeout(() => setFxState((prev) => ({ ...prev, [ch]: null })), 420);
   265→   265→    timersRef.current.push(id);
   266→   266→  }, []);
   267→   267→
   268→   268→  /* ---------- response evaluation (independent channels) ---------- */
   269→   269→  const press = useCallback((ch: Channel) => {
   270→   270→    if (phaseRef.current !== 'playing') return;
   271→   271→    const i = trialRef.current;
   272→   272→    const n = nRef.current;
   273→   273→    if (i < n) return; // warm-up trials have no n-back comparison yet — ignore early presses
   274→   274→    if (respRef.current[ch]) return; // one response per channel per window
   275→   275→    respRef.current[ch] = true;
   276→   276→    const seq = seqRef.current;
   277→   277→    const isMatch = ch === 'pos' ? seq.pos[i] === seq.pos[i - n] : seq.snd[i] === seq.snd[i - n];
   278→   278→    const st = statsRef.current[ch];
   279→   279→    if (isMatch) {
   280→   280→      st.hits++;
   281→   281→      sessionRef.current.points += HIT_POINTS * n;
   282→   282→      playBlip('good', soundRef.current);
   283→   283→      setFx(ch, 'hit');
   284→   284→    } else {
   285→   285→      st.fa++;
   286→   286→      sessionRef.current.points -= FA_POINTS * n;
   287→   287→      playBlip('bad', soundRef.current);
   288→   288→      setFx(ch, 'fa');
   289→   289→    }
   290→   290→    setScore(Math.max(0, sessionRef.current.points));
   291→   291→  }, [setFx]);
   292→   292→
   293→   293→  useEffect(() => { pressRef.current = press; }, [press]);
   294→   294→
   295→   295→  /* keyboard: A = position match, L = sound match (layout-independent via e.code) */
   296→   296→  useEffect(() => {
   297→   297→    const onKey = (e: KeyboardEvent) => {
   298→   298→      if (e.repeat) return;
   299→   299→      if (phaseRef.current !== 'playing') return;
   300→   300→      if (e.code === 'KeyA') {
   301→   301→        e.preventDefault();
   302→   302→        pressRef.current('pos');
   303→   303→      } else if (e.code === 'KeyL') {
   304→   304→        e.preventDefault();
   305→   305→        pressRef.current('snd');
   306→   306→      }
   307→   307→    };
   308→   308→    window.addEventListener('keydown', onKey);
   309→   309→    return () => window.removeEventListener('keydown', onKey);
   310→   310→  }, []);
   311→   311→
   312→   312→  /* unmount: kill every timer + close the audio graph */
   313→   313→  useEffect(() => {
   314→   314→    return () => {
   315→   315→      for (const id of timersRef.current) window.clearTimeout(id);
   316→   316→      timersRef.current = [];
   317→   317→      releaseAudio();
   318→   318→    };
   319→   319→  }, []);
   320→   320→
   321→   321→  /* campaign progress from localStorage (deferred: avoids sync setState-in-effect + SSR mismatch) */
   322→   322→  useEffect(() => {
   323→   323→    const id = window.setTimeout(() => setMaxDay(loadMaxDay()), 0);
   324→   324→    return () => window.clearTimeout(id);
   325→   325→  }, []);
   326→   326→
   327→   327→  /* ---------- trial engine ---------- */
   328→   328→
   329→   329→  /** Window for trial i has ended: register misses on the opportunities. */
   330→   330→  const commitTrial = (i: number) => {
   331→   331→    if (i < 0) return;
   332→   332→    const n = nRef.current;
   333→   333→    if (i < n) return; // warm-up trials carry no opportunities
   334→   334→    const st = statsRef.current;
   335→   335→    const seq = seqRef.current;
   336→   336→    if (seq.pos[i] === seq.pos[i - n] && !respRef.current.pos) st.pos.misses++;
   337→   337→    if (seq.snd[i] === seq.snd[i - n] && !respRef.current.snd) st.snd.misses++;
   338→   338→  };
   339→   339→
   340→   340→  const finishSession = () => {
   341→   341→    const s = sessionRef.current;
   342→   342→    const pos = sumStats(s.blocks.map((b) => b.pos));
   343→   343→    const snd = sumStats(s.blocks.map((b) => b.snd));
   344→   344→    const hits = pos.hits + snd.hits;
   345→   345→    const opps = pos.opps + snd.opps;
   346→   346→    const fas = pos.fa + snd.fa;
   347→   347→    const misses = pos.misses + snd.misses;
   348→   348→    const finalScore = Math.max(0, s.points);
   349→   349→    const accPct = opps > 0 ? Math.round((hits / opps) * 100) : 0;
   350→   350→    setScore(finalScore);
   351→   351→    setSummary({ title: s.title, blocks: [...s.blocks], pos, snd, score: finalScore, accPct });
   352→   352→    if (s.mode === 'campaign' && s.day > 0) {
   353→   353→      const prev = loadMaxDay();
   354→   354→      if (s.day > prev) {
   355→   355→        saveMaxDay(s.day);
   356→   356→        setMaxDay(s.day);
   357→   357→      }
   358→   358→    }
   359→   359→    go('results');
   360→   360→    // Economy contract — guests fail gracefully; local results still shown.
   361→   361→    void finish(finalScore, {
   362→   362→      day: s.mode === 'campaign' ? s.day : 0,
   363→   363→      n: s.blocks.length > 0 ? s.blocks[s.blocks.length - 1].n : s.startN,
   364→   364→      hits,
   365→   365→      misses,
   366→   366→      falseAlarms: fas,
   367→   367→      accuracy: accPct,
   368→   368→    });
   369→   369→  };
   370→   370→
   371→   371→  const endBlock = () => {
   372→   372→    const n = nRef.current;
   373→   373→    const st = statsRef.current;
   374→   374→    const hits = st.pos.hits + st.snd.hits;
   375→   375→    const opps = st.pos.opps + st.snd.opps;
   376→   376→    const fas = st.pos.fa + st.snd.fa;
   377→   377→    const acc = effectiveAcc(hits, fas, opps);
   378→   378→    sessionRef.current.blocks.push({ n, pos: { ...st.pos }, snd: { ...st.snd }, acc });
   379→   379→    saveLastN(n);
   380→   380→    if (blockNumRef.current < SESSION_BLOCKS) {
   381→   381→      const nn = nextN(n, acc);
   382→   382→      setInterlude({ block: blockNumRef.current, prevN: n, nextNV: nn, acc, hits, fas });
   383→   383→      nRef.current = nn;
