   125→   125→function angLerp(a: number, b: number, k: number) {
   126→   126→  const d = ((b - a + Math.PI * 3) % (Math.PI * 2)) - Math.PI;
   127→   127→  return a + d * k;
   128→   128→}

   130→   130→/** Circle vs axis-aligned rect: push-out normal + penetration (null = miss). */
   131→   131→function circleRect(px: number, py: number, r: number, w: MMWall) {
   132→   132→  const cx = clamp(px, w.x, w.x + w.w);
   133→   133→  const cy = clamp(py, w.y, w.y + w.h);
   134→   134→  const dx = px - cx;
   135→   135→  const dy = py - cy;
   136→   136→  const d2 = dx * dx + dy * dy;
   137→   137→  if (d2 >= r * r) return null;
   138→   138→  if (d2 > 1e-6) {
   139→   139→    const d = Math.sqrt(d2);
   140→   140→    return { nx: dx / d, ny: dy / d, pen: r - d };
   141→   141→  }
   142→   142→  const l = px - w.x;
   143→   143→  const rt = w.x + w.w - px;
   144→   144→  const t = py - w.y;
   145→   145→  const b = w.y + w.h - py;
   146→   146→  const m = Math.min(l, rt, t, b);
   147→   147→  if (m === l) return { nx: -1, ny: 0, pen: r + l };
   148→   148→  if (m === rt) return { nx: 1, ny: 0, pen: r + rt };
   149→   149→  if (m === t) return { nx: 0, ny: -1, pen: r + t };
   150→   150→  return { nx: 0, ny: 1, pen: r + b };
   151→   151→}

   153→   153→/** Liang–Barsky segment vs AABB — used for drone line-of-sight. */
   154→   154→function segHitsRect(x1: number, y1: number, x2: number, y2: number, w: MMWall) {
   155→   155→  let t0 = 0;
   156→   156→  let t1 = 1;
   157→   157→  const dx = x2 - x1;
   158→   158→  const dy = y2 - y1;
   159→   159→  const p = [-dx, dx, -dy, dy];
   160→   160→  const q = [x1 - w.x, w.x + w.w - x1, y1 - w.y, w.y + w.h - y1];
   161→   161→  for (let i = 0; i < 4; i++) {
   162→   162→    if (p[i] === 0) {
   163→   163→      if (q[i] < 0) return false;
   164→   164→      continue;
   165→   165→    }
   166→   166→    const r = q[i] / p[i];
   167→   167→    if (p[i] < 0) {
   168→   168→      if (r > t1) return false;
   169→   169→      if (r > t0) t0 = r;
   170→   170→    } else {
   171→   171→      if (r < t0) return false;
   172→   172→      if (r < t1) t1 = r;
   173→   173→    }
   174→   174→  }
   175→   175→  return true;
   176→   176→}

   178→   178→function hasLOS(x1: number, y1: number, x2: number, y2: number, walls: MMWall[]) {
   179→   179→  for (const w of walls) if (segHitsRect(x1, y1, x2, y2, w)) return false;
   180→   180→  return true;
   181→   181→}

   183→   183→function pointFree(x: number, y: number, r: number, walls: MMWall[]) {
   184→   184→  for (const w of walls) if (circleRect(x, y, r, w)) return false;
   185→   185→  return true;
   186→   186→}

   188→   188→/** Push a circle body out of every wall; damp velocity into surfaces (e = bounce). */
   189→   189→function resolveWalls(x: number, y: number, r: number, vx: number, vy: number, e: number, walls: MMWall[]) {
   190→   190→  for (const w of walls) {
   191→   191→    const c = circleRect(x, y, r, w);
   192→   192→    if (!c) continue;
   193→   193→    x += c.nx * c.pen;
   194→   194→    y += c.ny * c.pen;
   195→   195→    const vn = vx * c.nx + vy * c.ny;
   196→   196→    if (vn < 0) {
   197→   197→      vx -= (1 + e) * vn * c.nx;
   198→   198→      vy -= (1 + e) * vn * c.ny;
   199→   199→    }
   200→   200→  }
   201→   201→  return { x, y, vx, vy };
   202→   202→}
   203→   203→
   204→   204→/** Keep a body inside the arena frame; e = 1 gives a clean pulse bounce. */
   205→   205→function resolveBounds(x: number, y: number, r: number, vx: number, vy: number, e: number) {
   206→   206→  let hit = false;
   207→   207→  if (x < PAD + r) { x = PAD + r; if (vx < 0) { vx = -vx * e; hit = true; } }
   208→   208→  if (x > ARENA_W - PAD - r) { x = ARENA_W - PAD - r; if (vx > 0) { vx = -vx * e; hit = true; } }
   209→   209→  if (y < PAD + r) { y = PAD + r; if (vy < 0) { vy = -vy * e; hit = true; } }
   210→   210→  if (y > ARENA_H - PAD - r) { y = ARENA_H - PAD - r; if (vy > 0) { vy = -vy * e; hit = true; } }
   211→   211→  return { x, y, vx, vy, hit };
   212→   212→}

   214→   214→/** Intercept angle at a moving target for a projectile of `speed` (per frame). */
   215→   215→function interceptAngle(sx: number, sy: number, tx: number, ty: number, tvx: number, tvy: number, speed: number) {
   216→   216→  const dx = tx - sx;
   217→   217→  const dy = ty - sy;
   218→   218→  const a = tvx * tvx + tvy * tvy - speed * speed;
   219→   219→  const b = 2 * (dx * tvx + dy * tvy);
   220→   220→  const c = dx * dx + dy * dy;
   221→   221→  let t = -1;
   222→   222→  if (Math.abs(a) < 1e-4) {
   223→   223→    if (Math.abs(b) > 1e-4) t = -c / b;
   224→   224→  } else {
   225→   225→    const disc = b * b - 4 * a * c;
   226→   226→    if (disc >= 0) {
   227→   227→      const sq = Math.sqrt(disc);
   228→   228→      const t1 = (-b - sq) / (2 * a);
   229→   229→      const t2 = (-b + sq) / (2 * a);
   230→   230→      t = Math.min(t1, t2);
   231→   231→      if (t < 0) t = Math.max(t1, t2);
   232→   232→    }
   233→   233→  }
   234→   234→  if (!Number.isFinite(t) || t < 0) t = Math.hypot(dx, dy) / speed;
   235→   235→  t = Math.min(t, 400); // sanity bound; real intercepts are far below this
   236→   236→  return Math.atan2(dy + tvy * t, dx + tvx * t);
   237→   237→}

   239→   239→function rr(ctx: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, r: number) {
   240→   240→  const rad = Math.min(r, w / 2, h / 2);
   241→   241→  ctx.beginPath();
   242→   242→  ctx.moveTo(x + rad, y);
   243→   243→  ctx.arcTo(x + w, y, x + w, y + h, rad);
   244→   244→  ctx.arcTo(x + w, y + h, x, y + h, rad);
   245→   245→  ctx.arcTo(x, y + h, x, y, rad);
   246→   246→  ctx.arcTo(x, y, x + w, y, rad);
   247→   247→  ctx.closePath();
   248→   248→}
