    83→/** 'piece_rose' → 'Rose', 'bg_royal_velvet' → 'Royal velvet' */
    84→function prettyName(code: string): string {
    85→  const s = code.split('_').slice(1).join(' ').trim();
    86→  return s ? s.charAt(0).toUpperCase() + s.slice(1) : code;
    87→}

    89→/** Immutably write the equipped code into the right skin/avatar/title field. */
    90→function equipApply<T extends { pieceSkin: string; boardSkin: string; background: string; effectSkin: string; avatarItem: string; titleItem: string | null }>(
    91→  obj: T,
    92→  field: EquipField,
    93→  code: string,
    94→): T {
    95→  const next = { ...obj };
    96→  if (field === 'pieceSkin') next.pieceSkin = code;
    97→  else if (field === 'boardSkin') next.boardSkin = code;
    98→  else if (field === 'background') next.background = code;
    99→  else if (field === 'effectSkin') next.effectSkin = code;
   100→  else if (field === 'avatarItem') next.avatarItem = code;
   101→  else next.titleItem = code;
   102→  return next;
   103→}

   107→function ItemPreview({ item, glow }: { item: StoreItemDTO; glow: boolean }) {
   108→  const d = item.data as { cell?: string; cell2?: string; hole?: string; numeral?: string; colors?: string[]; accent?: string };
   109→
   110→  switch (item.category) {
   111→    case 'PIECE':
   112→      return (
   113→        <div style={glow ? { filter: `drop-shadow(0 0 7px ${RARITY_META[item.rarity].glow})` } : undefined}>
   114→          <PieceDisc code={item.code} side={1} size={64} />
   115→        </div>
   116→      );
   117→    case 'AVATAR':
   118→      return <AvatarBubble code={item.code} size={64} glow={glow} />;
   119→    case 'BOARD': {
   120→      const cell = d.cell ?? '#e3b878';
   121→      const cell2 = d.cell2 ?? '#d9a860';
   122→      const hole = d.hole ?? '#faf6ee';
   123→      const numeral = d.numeral ?? '#4a2e12';
   124→      return (
   125→        <div
   126→          className="grid h-16 w-16 place-items-center rounded-lg"
   127→          style={{
   128→            background: `linear-gradient(135deg, ${cell}, ${cell2})`,
   129→            border: `2px solid ${numeral}`,
   130→            boxShadow: 'inset 0 1px 5px rgba(0,0,0,.35)',
   131→          }}
   132→        >
   133→          <span className="block h-3.5 w-3.5 rounded-full" style={{ background: hole, boxShadow: `0 0 6px ${numeral}` }} />
   134→        </div>
   135→      );
   136→    }
   137→    case 'BACKGROUND': {
   138→      const c = d.colors ?? ['#0f0524', '#1e1145', '#2a1a5e'];
   139→      const accent = d.accent ?? '#d4af37';
   140→      return (
   141→        <div
   142→          className="relative h-16 w-16 overflow-hidden rounded-lg border border-white/15"
   143→          style={{ background: `linear-gradient(135deg, ${c[0] ?? '#000'}, ${c[1] ?? '#111'} 55%, ${c[2] ?? '#222'})` }}

   156→          className="grid h-16 w-16 place-items-center text-[34px]"

   165→          🎉

   176→        </div>

   179→      return null;
   180→  }

   310→const CONFETTI_COLORS = ['#fbbf24', '#f472b6', '#34d399', '#c084fc', '#f43f5e', '#38bdf8', '#fde047'];
   311→
   312→/** Lightweight confetti — absolutely-positioned animated divs, no new deps. */
   313→function Confetti({ seed }: { seed: number }) {
   314→  const parts = useMemo(
   315→    () =>
   316→      Array.from({ length: 30 }, (_, i) => ({
   317→        id: i,
   318→        x: (Math.random() - 0.5) * 340,
   319→        y: -(70 + Math.random() * 170),
   320→        rot: Math.random() * 720 - 360,
   321→        dur: 1 + Math.random(),
   322→        delay: Math.random() * 0.18,
   323→        c: CONFETTI_COLORS[i % CONFETTI_COLORS.length],
   324→        w: 5 + Math.random() * 6,
   325→      })),
   326→    [seed],
   327→  );
   328→  return (
   329→    <div className="pointer-events-none absolute inset-0 overflow-hidden" aria-hidden>
   330→      {parts.map((p) => (
   331→        <motion.span
   332→          key={`${seed}-${p.id}`}
   333→          className="absolute left-1/2 top-[38%] block rounded-[2px]"
   334→          style={{ width: p.w, height: Math.round(p.w * 0.5), background: p.c }}
   335→          initial={{ x: 0, y: 0, opacity: 1, rotate: 0 }}
   336→          animate={{ x: p.x, y: [0, p.y, 330], opacity: [1, 1, 0], rotate: p.rot }}
   337→          transition={{ duration: p.dur + 0.9, delay: p.delay, ease: 'easeOut', times: [0, 0.42, 1] }}
   338→        />
   339→      ))}
   340→    </div>
   341→  );
   342→}

   344→/** Big reward icon inside the reveal ring (reuses the item renderers above). */
   345→function RewardVisual({ kind, itemCode, catalogItem }: { kind: DropKind; itemCode: string | null; catalogItem: StoreItemDTO | null }) {
   346→  if (kind === 'COINS') return <span className="text-5xl drop-shadow-[0_0_16px_rgba(251,191,36,.85)]" aria-hidden>🪙</span>;
   347→  if (kind === 'XP') return <span className="text-5xl text-violet-300 drop-shadow-[0_0_16px_rgba(192,132,252,.85)]" aria-hidden>✦</span>;
   348→  const code = itemCode ?? '';
   349→  const prefix = code.split('_')[0] ?? '';
   350→  if (prefix === 'piece') return <PieceDisc code={code} side={1} size={72} />;
   351→  if (prefix === 'avatar') return <AvatarBubble code={code} size={72} glow />;
   352→  if (catalogItem) return <ItemPreview item={catalogItem} glow />;
   353→  return <span className="text-5xl" aria-hidden>{ITEM_EMOJI[prefix] ?? '📦'}</span>;
   354→}
