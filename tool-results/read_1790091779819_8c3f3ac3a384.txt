     1→/**
     2→ * Game content catalog: store items (skins/avatars/backgrounds/titles/effects),
     3→ * achievements, daily quests, rarity tiers. Item `data` JSON drives the
     4→ * procedural pixel-art renderer (client) so no binary art assets are needed.
     5→ */
     6→
     7→export type Rarity = 'COMMON' | 'RARE' | 'EPIC' | 'LEGENDARY' | 'ULTIMATE';
     8→export type ItemCategory = 'PIECE' | 'BOARD' | 'AVATAR' | 'BACKGROUND' | 'EFFECT' | 'TITLE' | 'EMOTE';
     9→
    10→export interface ItemDef {
    11→  code: string;
    12→  category: ItemCategory;
    13→  rarity: Rarity;
    14→  price: number;
    15→  requiredLevel: number;
    16→  data: Record<string, unknown>;
    17→}

    27→/** Every piece skin defines palettes for BOTH sides + a pixel pattern seed. */
    28→const piece = (code: string, rarity: Rarity, price: number, p1: string[], p2: string[], seed: number, nameGlow?: string): ItemDef =>
    29→  ({ code, category: 'PIECE', rarity, price, requiredLevel: rarity === 'ULTIMATE' ? 12 : rarity === 'LEGENDARY' ? 6 : rarity === 'EPIC' ? 3 : 1, data: { p1, p2, seed, glow: nameGlow ?? p1[0] } });
    30→
    31→const avatar = (code: string, rarity: Rarity, price: number, seed: number, pal: string[]): ItemDef =>
    32→  ({ code, category: 'AVATAR', rarity, price, requiredLevel: 1, data: { seed, pal } });
    33→
    34→const bg = (code: string, rarity: Rarity, price: number, kind: string, colors: string[], accent: string): ItemDef =>
    35→  ({ code, category: 'BACKGROUND', rarity, price, requiredLevel: 1, data: { kind, colors, accent } });
    36→
    37→const board = (code: string, rarity: Rarity, price: number, pal: Record<string, string>): ItemDef =>
    38→  ({ code, category: 'BOARD', rarity, price, requiredLevel: rarity === 'ULTIMATE' ? 12 : rarity === 'EPIC' ? 3 : 1, data: pal });
    39→
    40→const fx = (code: string, rarity: Rarity, price: number, colors: string[], shape: string): ItemDef =>
    41→  ({ code, category: 'EFFECT', rarity, price, requiredLevel: 1, data: { colors, shape } });
    42→
    43→const title = (code: string, rarity: Rarity, price: number, reqLevel: number): ItemDef =>
    44→  ({ code, category: 'TITLE', rarity, price, requiredLevel: reqLevel, data: {} });

   249→/** Deterministically pick 3 quests per user per day (seeded by dateKey+userId). */
   250→export function questsForDay(dateKey: string, userId: string): QuestDef[] {
   251→  let seed = 2166136261;
   252→  const str = dateKey + '|' + userId;
   253→  for (let i = 0; i < str.length; i++) {
   254→    seed ^= str.charCodeAt(i);
   255→    seed = Math.imul(seed, 16777619);
   256→  }
   257→  const rnd = (() => { let a = seed >>> 0; return () => { a |= 0; a = (a + 0x6d2b79f5) | 0; let t = Math.imul(a ^ (a >>> 15), 1 | a); t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t; return ((t ^ (t >>> 14)) >>> 0) / 4294967296; }; })();
   258→  const pool = [...QUEST_POOL];
   259→  const picked: QuestDef[] = [];
   260→  while (picked.length < 3 && pool.length) {
   261→    picked.push(pool.splice(Math.floor(rnd() * pool.length), 1)[0]);
   262→  }
   263→  return picked;
   264→}

   267→export function xpForLevel(level: number): number {
   268→  // XP needed to advance FROM `level` to level+1
   269→  return 100 + (level - 1) * 60;
   270→}

   272→export function levelFromXp(xp: number): { level: number; into: number; need: number } {
   273→  let level = 1;
   274→  let remaining = xp;
   275→  while (remaining >= xpForLevel(level) && level < 99) {
   276→    remaining -= xpForLevel(level);
   277→    level++;
   278→  }
   279→  return { level, into: remaining, need: xpForLevel(level) };
   280→}

   282→export interface RewardInput {
   283→  mode: 'PVE' | 'LOCAL_PVP' | 'ONLINE';
   284→  outcome: 'WIN' | 'LOSS' | 'DRAW';
   285→  aiLevel?: 'EASY' | 'MEDIUM' | 'HARD';
   286→  ranked?: boolean;
   287→  moves: number;
   288→  streak: number; // current streak BEFORE this game (for win bonus)
   289→}
   290→
   291→export function computeRewards(r: RewardInput): { coins: number; xp: number } {
   292→  let coins = 0;
   293→  let xp = 0;
   294→  if (r.mode === 'ONLINE') {
   295→    if (r.outcome === 'WIN') { coins = 60 + Math.min(r.streak, 5) * 6; xp = 80; }
   296→    else if (r.outcome === 'DRAW') { coins = 28; xp = 40; }
   297→    else { coins = 12; xp = 20; }
   298→  } else if (r.mode === 'PVE') {
   299→    const base = r.aiLevel === 'HARD' ? 45 : r.aiLevel === 'MEDIUM' ? 22 : 8;
   300→    if (r.outcome === 'WIN') { coins = base; xp = 30 + (r.aiLevel === 'HARD' ? 30 : r.aiLevel === 'MEDIUM' ? 12 : 0); }
   301→    else if (r.outcome === 'DRAW') { coins = Math.ceil(base / 3); xp = 15; }
   302→    else { coins = Math.ceil(base / 5); xp = 8; }
   303→  } else { // LOCAL_PVP — fun + XP only (anti-farm: near-zero coins)
   304→    coins = 4; xp = 25;
   305→  }
   306→  return { coins, xp };
   307→}

   309→/** ELO with K=32; returns deltas from p1 perspective */
   310→export function eloDeltas(r1: number, r2: number, p1Score: 0 | 0.5 | 1, k = 32): [number, number] {
   311→  const exp = 1 / (1 + Math.pow(10, (r2 - r1) / 400));
   312→  const d = Math.round(k * (p1Score - exp));
   313→  return [d, -d];
   314→}

   402→/** Weighted random pick (server-side). Returns the winning drop. */
   403→export function rollDrop(drops: DropDef[], rnd: () => number = Math.random): DropDef {
   404→  const total = drops.reduce((s, d) => s + d.weight, 0);
   405→  let r = rnd() * total;
   406→  for (const d of drops) {
   407→    r -= d.weight;
   408→    if (r <= 0) return d;
   409→  }
   410→  return drops[drops.length - 1];
   411→}

   414→export type ArcadeGame = 'orbit' | 'synapse' | 'stellar' | 'mazemind' | 'boulderyard' | 'boulderyard2';
   415→export const ARCADE_GAMES: ArcadeGame[] = ['orbit', 'synapse', 'stellar', 'mazemind', 'boulderyard', 'boulderyard2'];
   416→
   417→/** Per-game hourly coin caps (anti-farm, mirrors duel CAPS). */
   418→export const ARCADE_CAPS: Record<ArcadeGame, number> = { orbit: 260, synapse: 220, stellar: 240, mazemind: 300, boulderyard: 260, boulderyard2: 280 };
   419→
   420→/**
   421→ * Score -> rewards. Score scales differ per game; each formula is tuned so a
   422→ * solid mid-tier run ≈ a duel win. Server clamps by hourly cap.
   423→ */
   424→export function arcadeRewards(game: ArcadeGame, score: number): { coins: number; xp: number } {
   425→  if (!Number.isFinite(score) || score <= 0) return { coins: 3, xp: 6 };
   426→  const s = Math.floor(score);
   427→  if (game === 'orbit') {
   428→    // perfect mission ≈ 1000; typical early mission ≈ 300
   429→    const coins = Math.min(90, 12 + Math.floor(s / 12));
   430→    const xp = Math.min(120, 20 + Math.floor(s / 9));
   431→    return { coins, xp };
   432→  }
   433→  if (game === 'synapse') {
   434→    // session score ≈ trials × accuracy × n
   435→    const coins = Math.min(80, 10 + Math.floor(s / 14));
   436→    const xp = Math.min(110, 18 + Math.floor(s / 10));
   437→    return { coins, xp };
   438→  }
   439→  // stellar: mass units forged; a full supernova run ≈ 2000+
   440→  const coins = Math.min(85, 10 + Math.floor(s / 24));
   441→  const xp = Math.min(115, 16 + Math.floor(s / 18));
   442→  return { coins, xp };
   443→}

   445→/** v0.2 reward curves for the three new arcade games (mind-tank wave scores,
   446→ *  boulder yard level scores). Same tuning goal: solid run ≈ a duel win. */
   447→export function arcadeRewardsV2(game: 'mazemind' | 'boulderyard' | 'boulderyard2', score: number): { coins: number; xp: number } {
   448→  if (!Number.isFinite(score) || score <= 0) return { coins: 3, xp: 6 };
   449→  const s = Math.floor(score);
   450→  if (game === 'mazemind') {
   451→    // a full 10-robot run ≈ 3200; casual early death ≈ 500
   452→    const coins = Math.min(100, 12 + Math.floor(s / 26));
   453→    const xp = Math.min(130, 16 + Math.floor(s / 20));
   454→    return { coins, xp };
   455→  }
   456→  if (game === 'boulderyard') {
   457→    // solving ~6 levels ≈ 1400
   458→    const coins = Math.min(90, 10 + Math.floor(s / 18));
   459→    const xp = Math.min(110, 14 + Math.floor(s / 15));
   460→    return { coins, xp };
   461→  }
   462→  // boulderyard2: trickier mechanics, slightly richer
   463→  const coins = Math.min(95, 10 + Math.floor(s / 17));
   464→  const xp = Math.min(120, 14 + Math.floor(s / 14));
   465→  return { coins, xp };
   466→}
