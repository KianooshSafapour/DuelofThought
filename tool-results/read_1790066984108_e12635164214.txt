   122→   122→/** 'first_blood' → 'First blood' (fallback for titles) */
   123→   123→function prettyName(code: string): string {
   124→   124→  const s = code.split('_').slice(1).join(' ').trim();
   125→   125→  return s ? s.charAt(0).toUpperCase() + s.slice(1) : code;
   126→   126→}

   128→   128→/** Localized achievement label built from its metric + goal (readable in all 4 locales). */
   129→   129→function achLabel(a: { id: string; metric: string; goal: number }, t: (k: TKey) => string): string {
   130→   130→  const g = a.goal;
   131→   131→  switch (a.metric) {
   132→   132→    case 'wins': return `${g} × ${t('wins')}`;
   133→   133→    case 'kindWins': return `${g > 1 ? `${g} × ` : ''}${t('quest_q_kind')}`;
   134→   134→    case 'runWins': return `${g > 1 ? `${g} × ` : ''}${t('quest_q_run')}`;
   135→   135→    case 'games': return `${g} × ${t('games_played')}`;
   136→   136→    case 'onlineWins': return `${g} × ${t('online')}`;
   137→   137→    case 'onlineGames': return `${g} × ${t('online')}`;
   138→   138→    case 'hardWins': return `${g} × AI ${t('ai_hard')}`;
   139→   139→    case 'streakBest': return `${g} × ${t('streak')}`;
   140→   140→    case 'itemsOwned': return `${g} × 🛍`;
   141→   141→    case 'level': return `${t('level')} ${g}`;
   142→   142→    case 'questsDone': return `${g} × ${t('daily_quests')}`;
   143→   143→    case 'draws': return `${g} × ${t('draws_stat')}`;
   144→   144→    default: return prettyName(a.id);
   145→   145→  }
   146→   146→}

   148→   148→/** Resolve an achievement id to its catalog def → localized label. */
   149→   149→function achLabelById(id: string, t: (k: TKey) => string): string {
   150→   150→  const def = ACHIEVEMENTS.find((x) => x.id === id);
   151→   151→  return def ? achLabel(def, t) : prettyName(id);
   152→   152→}

   154→   154→function fmtDate(iso: string, locale: Locale): string {
   155→   155→  try {
   156→   156→    return new Intl.DateTimeFormat(locale, { year: 'numeric', month: 'short', day: 'numeric' }).format(new Date(iso));
   157→   157→  } catch {
   158→   158→    return iso.slice(0, 10);
   159→   159→  }
   160→   160→}
   161→   161→
   162→   162→function timeAgo(iso: string, locale: Locale): string {
   163→   163→  try {
   164→   164→    const rtf = new Intl.RelativeTimeFormat(locale, { numeric: 'auto' });
   165→   165→    const s = Math.round((Date.now() - new Date(iso).getTime()) / 1000);
   166→   166→    if (s < 60) return rtf.format(-Math.max(s, 1), 'second');
   167→   167→    const m = Math.round(s / 60);
   168→   168→    if (m < 60) return rtf.format(-m, 'minute');
   169→   169→    const h = Math.round(m / 60);
   170→   170→    if (h < 24) return rtf.format(-h, 'hour');
   171→   171→    return rtf.format(-Math.round(h / 24), 'day');
   172→   172→  } catch {
   173→   173→    return '';
   174→   174→  }
   175→   175→}
